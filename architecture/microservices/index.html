Декомпозиция сервисов, API Gateway, паттерны коммуникации, деплой

Обзор: монолит → микросервисы

Было (монолит):
  FastAPI App ────→ PostgreSQL
       └──────────→ Keycloak
       └──────────→ Redis
  Всё в одном процессе → сложно масштабировать, один деплой на всё

Стало (микросервисы):
  Browser/App
       │
  [Traefik API Gateway]
       ├──→ tenantpass/iam-service     (Keycloak обёртка)
       ├──→ tenantpass/access-service  (СКУД PEP/PDP)
       ├──→ tenantpass/workforce-service (СУРВ, табели)
       ├──→ eventpass/event-service    (мероприятия)
       ├──→ eventpass/ticket-service   (QR-билеты)
       └──→ shared/ml-service          (Triton + Risk API)
  
  Сервисы общаются через:
    Sync:  gRPC (внутренние, low-latency)
    Async: Redpanda topics (события)
    IoT:   MQTT broker (устройства)

# REST: внешние клиенты и cross-domain вызовы
# access-service -> ml-service: получить risk score перед принятием решения
import httpx

async def get_risk_score(user_id: str, zone_id: str) -> float:
    async with httpx.AsyncClient(timeout=50) as c:
        r = await c.post("http://ml-service:8010/risk_score",
                         json={"user_id": user_id, "zone_id": zone_id})
    return r.json()["score"]

Circuit Breaker — защита от каскадных отказов

# Паттерн Circuit Breaker для межсервисных вызовов
# Используем tenacity + кастомный circuit breaker

import asyncio
from datetime import datetime, timedelta
from enum import Enum

class CircuitState(Enum):
    CLOSED = "closed"       # нормально, запросы проходят
    OPEN = "open"           # сервис упал, запросы блокируются
    HALF_OPEN = "half_open" # пробуем восстановиться

class CircuitBreaker:
    def __init__(self, failure_threshold: int = 5,
                 timeout: float = 60.0,
                 success_threshold: int = 2):
        self.state = CircuitState.CLOSED
        self.failures = 0
        self.last_failure_time: datetime | None = None
        self.success_count = 0
        self.failure_threshold = failure_threshold
        self.timeout = timeout
        self.success_threshold = success_threshold

    async def call(self, func, *args, **kwargs):
        if self.state == CircuitState.OPEN:
            elapsed = (datetime.now() - self.last_failure_time).total_seconds()
            if elapsed > self.timeout:
                self.state = CircuitState.HALF_OPEN
            else:
                raise Exception("Circuit OPEN — сервис недоступен")

        try:
            result = await func(*args, **kwargs)
            self._on_success()
            return result
        except Exception as e:
            self._on_failure()
            raise

    def _on_success(self):
        if self.state == CircuitState.HALF_OPEN:
            self.success_count += 1
            if self.success_count >= self.success_threshold:
                self.state = CircuitState.CLOSED
                self.failures = 0
        else:
            self.failures = 0

    def _on_failure(self):
        self.failures += 1
        self.last_failure_time = datetime.now()
        if self.failures >= self.failure_threshold:
            self.state = CircuitState.OPEN

# Использование: access-service → ml-service
ml_circuit = CircuitBreaker(failure_threshold=5, timeout=30.0)

async def safe_get_risk_score(user_id: str) -> float:
    try:
        return await ml_circuit.call(get_risk_score, user_id)
    except Exception:
        return 0.3  # fallback: нейтральный risk score

Health Checks & Service Discovery

# Каждый сервис реализует /health endpoint
# FastAPI пример для access-service

from fastapi import FastAPI
from pydantic import BaseModel

app = FastAPI()

class HealthStatus(BaseModel):
    status: str
    version: str
    dependencies: dict[str, str]

@app.get("/health")
async def health_check() -> HealthStatus:
    deps = {}
    # Проверка PostgreSQL
    try:
        await db.execute("SELECT 1")
        deps["postgres"] = "ok"
    except Exception:
        deps["postgres"] = "error"
    # Проверка Redis
    try:
        await redis.ping()
        deps["redis"] = "ok"
    except Exception:
        deps["redis"] = "error"
    # Проверка MQTT
    try:
        await mqtt.is_connected()
        deps["mqtt"] = "ok"
    except Exception:
        deps["mqtt"] = "error"

    all_ok = all(v == "ok" for v in deps.values())
    return HealthStatus(
        status="healthy" if all_ok else "degraded",
        version="1.2.0",
        dependencies=deps
    )
