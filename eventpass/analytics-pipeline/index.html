PostgreSQL → ClickHouse через Debezium CDC

Debezium читает PostgreSQL WAL (Write-Ahead Log) — без нагрузки на основной API.

ClickHouse: схема таблиц

-- ClickHouse: аналитические таблицы

CREATE TABLE events_raw (
    event_id     String,
    event_type   LowCardinality(String),   -- 'scan', 'grant', 'deny', 'purchase'
    event_ts     DateTime64(3, 'UTC'),
    event_date   Date MATERIALIZED toDate(event_ts),

    tenant_id    String,
    event_id_fk  String,
    zone_id      LowCardinality(String),
    scanner_id   String,
    person_id    String,
    decision     LowCardinality(String),   -- 'granted', 'denied', 'expired', 'replay'

    meta         String                    -- JSON строка для дополнительных полей
) ENGINE = MergeTree()
  PARTITION BY toYYYYMM(event_date)
  ORDER BY (tenant_id, event_date, event_type)
  TTL event_date + INTERVAL 2 YEAR;

-- Материализованное представление: почасовая статистика
CREATE MATERIALIZED VIEW mv_hourly_stats
ENGINE = SummingMergeTree()
PARTITION BY toYYYYMM(hour)
ORDER BY (tenant_id, event_id_fk, zone_id, hour)
AS SELECT
    tenant_id,
    event_id_fk,
    zone_id,
    toStartOfHour(event_ts)      AS hour,
    countIf(decision='granted')  AS entries,
    countIf(decision='denied')   AS denials,
    uniqExact(person_id)         AS unique_persons
FROM events_raw
GROUP BY tenant_id, event_id_fk, zone_id, hour;

Kafka → ClickHouse consumer

# analytics/ch_consumer.py
from confluent_kafka import Consumer
import clickhouse_connect, json

consumer = Consumer({
    "bootstrap.servers": "redpanda:9092",
    "group.id": "ch-sink",
    "auto.offset.reset": "earliest",
})
consumer.subscribe(["eventpass.public.scan_events"])

ch = clickhouse_connect.get_client(host="clickhouse", port=8123)

buffer = []
BATCH_SIZE = 1000

while True:
    msg = consumer.poll(timeout=1.0)
    if msg and not msg.error():
        payload = json.loads(msg.value())["payload"]
        if payload["op"] in ("c", "u"):   # create or update
            after = payload["after"]
            buffer.append([
                after["id"], "scan",
                after["scan_ts"], after["tenant_id"],
                after["event_id"], after.get("zone_id", ""),
                after["scanner_id"], after.get("person_id", ""),
                after["decision"], json.dumps(after.get("meta", {}))
            ])

    if len(buffer) >= BATCH_SIZE:
        ch.insert("events_raw",
                  buffer,
                  column_names=["event_id","event_type","event_ts","tenant_id",
                                "event_id_fk","zone_id","scanner_id","person_id",
                                "decision","meta"])
        buffer.clear()

Grafana дашборды

Ключевые дашборды (ClickHouse datasource):

1. Realtime: Посетители сейчас
   SELECT zone_id, entries FROM mv_hourly_stats
   WHERE hour = toStartOfHour(NOW()) AND event_id_fk = $event_id
   ORDER BY entries DESC

2. Поток входов по времени (15-мин интервалы)
   SELECT toStartOfFifteenMinutes(event_ts) AS t, count() AS cnt
   FROM events_raw WHERE event_date = today() AND decision='granted'
   GROUP BY t ORDER BY t

3. Топ причин отказов
   SELECT meta, count() FROM events_raw WHERE decision='denied'
   GROUP BY meta ORDER BY count() DESC LIMIT 10

4. Заполненность зон (heat map)
   → данные из mv_hourly_stats по зонам

Прямая запись (без Kafka) — простой вариант

Если масштаб мероприятия < 10 000 человек, Kafka избыточна. Достаточно прямой записи из Event API.

# Простой вариант: Event API пишет в ClickHouse напрямую
import clickhouse_connect

ch_client = clickhouse_connect.get_client(
    host=settings.CLICKHOUSE_HOST,
    port=8123,
    database="eventpass_analytics"
)

async def record_scan_event(scan: ScanEvent):
    # Писать в PostgreSQL (operational)
    await pg.execute("INSERT INTO scan_events ...", ...)

    # Писать в ClickHouse (analytics) — async, не блокирует ответ API
    asyncio.create_task(
        asyncio.to_thread(
            ch_client.insert,
            "events_raw",
            [scan.id, "scan", scan.ts, ...](/scan.id, "scan", scan.ts, .../),
            column_names=[...]
        )
    )
