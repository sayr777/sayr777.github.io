Offline-режим сканеров EventPass (замечания 7+8)

EventPass — автономная валидация QR при потере 4G

Проблема

ESP32+4G сканер → при потере связи → нет доступа к Event API → не может проверить QR → блокада на входе. На стройке или стадионе 4G может пропасть в любой момент.

Решение: JWT RS256 + offline-кэш

Вместо HMAC (symmetric) — ассиметричный JWT (RS256). Публичный ключ загружается в сканер при инициализации. Сканер валидирует подпись без обращения к серверу.

sequenceDiagram participant C as ESP32 Scanner participant S as Event API participant QR as QR-код (JWT) note over C,S: Инициализация (при наличии связи) C->>S: GET /api/v1/scanner/init S-->>C: {public_key_pem, event_id, valid_zones, revocation_list} C->>C: Сохранить в flash (SPIFFS) note over C,QR: Нормальная работа (с 4G) QR-->>C: scan QR → JWT token C->>C: verify JWT signature (public key) C->>C: check claims (event_id, zone, exp, jti) C->>S: POST /api/v1/scan {jti, scanner_id, ts} S-->>C: {valid, person_name, access_level} C->>C: пометить jti как использованный (local set) C->>C: sync used_jti to server note over C,QR: Offline режим (нет 4G) QR-->>C: scan QR → JWT token C->>C: verify JWT signature (public key из flash) C->>C: check exp, event_id, zone C->>C: check jti NOT IN local_used_set C->>C: ✅ Allow / ❌ Deny (без сервера) C->>C: queue scan event for later sync

Key rotation (замечание 8)

Каждое мероприятие — отдельная ключевая пара. Компрометация ключа одного события не затрагивает другие.

# eventpass/key_manager.py
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives import serialization
from datetime import datetime, timedelta
import uuid

class EventKeyManager:

    def create_event_keypair(self, event_id: str) -> dict:
        """Создать ключевую пару для мероприятия"""
        private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
        )
        kid = f"{event_id}-{uuid.uuid4().hex[:8]}"

        private_pem = private_key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=serialization.NoEncryption()
        ).decode()

        public_pem = private_key.public_key().private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PublicFormat.SubjectPublicKeyInfo,
        ).decode()

        # Сохранить в БД (приватный — зашифрованный, публичный — открыто)
        db.execute("""
            INSERT INTO event_keys (event_id, kid, public_key_pem, private_key_enc, created_at)
            VALUES ($1, $2, $3, encrypt($4, $5), NOW())
        """, event_id, kid, public_pem, private_pem, KEY_ENCRYPTION_SECRET)

        return {"kid": kid, "public_key_pem": public_pem}

    def rotate_key(self, event_id: str, reason: str) -> dict:
        """Экстренная ротация при подозрении на компрометацию"""
        # 1. Создать новую пару
        new_key = self.create_event_keypair(event_id)
        # 2. Старые ключи — деактивировать (токены с old kid → rejected)
        db.execute("UPDATE event_keys SET revoked_at=NOW(), revoke_reason=$1 WHERE event_id=$2 AND revoked_at IS NULL",
                   reason, event_id)
        # 3. Уведомить все сканеры — перезагрузить ключи
        scanner_notify.broadcast(event_id, {"action": "reload_key", "new_kid": new_key["kid"]})
        return new_key

ESP32: offline JWT validation

// src/jwt_validator.cpp
// Библиотека: arduino-jwt (или собственная RS256)

#include <ArduinoJson.h>
#include <mbedtls/rsa.h>
#include <mbedtls/sha256.h>

bool validateJWT(const String& token, const String& publicKeyPEM,
                 const String& expectedEventId, const String& scannerZone) {

    // 1. Разбить токен на 3 части
    int dot1 = token.indexOf('.');
    int dot2 = token.lastIndexOf('.');
    String header_b64  = token.substring(0, dot1);
    String payload_b64 = token.substring(dot1+1, dot2);
    String sig_b64     = token.substring(dot2+1);

    // 2. Проверить подпись RS256
    if (!verifyRS256Signature(header_b64 + "." + payload_b64, sig_b64, publicKeyPEM)) {
        Serial.println("JWT: invalid signature");
        return false;
    }

    // 3. Декодировать payload
    String payload_json = base64url_decode(payload_b64);
    DynamicJsonDocument doc(512);
    deserializeJson(doc, payload_json);

    // 4. Проверить exp
    long exp = doc["exp"];
    if (exp < getCurrentUnixTime()) {
        Serial.println("JWT: expired");
        return false;
    }

    // 5. Проверить event_id
    if (doc["event_id"].as<String>() != expectedEventId) {
        Serial.println("JWT: wrong event");
        return false;
    }

    // 6. Проверить zone
    JsonArray zones = doc["zone"].as<JsonArray>();
    bool zoneAllowed = false;
    for (JsonVariant z : zones) {
        if (z.as<String>() == scannerZone) { zoneAllowed = true; break; }
    }
    if (!zoneAllowed) {
        Serial.println("JWT: zone not allowed");
        return false;
    }

    // 7. Проверить одноразовость (jti в локальном наборе)
    String jti = doc["jti"].as<String>();
    if (usedJtiSet.contains(jti)) {
        Serial.println("JWT: already used (anti-replay)");
        return false;
    }
    usedJtiSet.insert(jti);   // Set<String> в PSRAM

    return true;
}

Синхронизация при восстановлении связи

# eventpass/scanner_sync.py
# ESP32 накапливает события → при появлении 4G отправляет пачкой

async def sync_offline_events(scanner_id: str, events: list[dict]):
    """
    events: [{jti, scan_ts, decision, person_name_from_jwt}]
    Сервер: записать в БД, обновить статистику, проверить replay
    """
    for event in sorted(events, key=lambda e: e["scan_ts"]):
        # Проверить: не было ли этот jti использован на другом сканере
        existing = await db.fetchval("SELECT id FROM scan_events WHERE jti=$1", event["jti"])
        if existing:
            # Replay attack: один билет — два входа
            await alert_security(scanner_id, event, "replay_detected")
        else:
            await db.execute("""
                INSERT INTO scan_events (jti, scanner_id, scan_ts, decision, synced_at)
                VALUES ($1, $2, $3, $4, NOW())
            """, event["jti"], scanner_id, event["scan_ts"], event["decision"])
