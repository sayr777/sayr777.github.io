Техзона: кабели и коммуникации

Чердак — узловая точка прокладки кабелей:
  Антенна GSM / WiFi-сектор (OpenWrt AP)
  PoE-свитч для IP-камер кровли
  Кросс-панель (витая пара к комнатам)
  ИК-ретранслятор для управления кондиционерами
  Трубы вентиляции (выходят через кровлю)

Сенсоры чердака

# esphome/attic.yaml
sensor:
  # Температура (DS18B20 под коньком — критично летом)
  - platform: dallas_temp
    address: 0x283c01d607923c32
    name: "Чердак — температура"
    id: attic_temp

  # Влажность (BME280 — риск конденсата и грибка)
  - platform: bme280_i2c
    temperature:
      name: "Мансарда — температура"
    humidity:
      name: "Мансарда — влажность"
      id: attic_humidity
    address: 0x76

binary_sensor:
  # Движение (охрана, если чердак жилой)
  - platform: gpio
    name: "Чердак: движение"
    device_class: motion
    pin: GPIO33

  # Люк открыт/закрыт
  - platform: gpio
    name: "Люк чердака: открыт"
    device_class: door
    pin:
      number: GPIO34
      mode: INPUT_PULLUP
      inverted: true

Солнечные панели (на кровле)

# Мониторинг солнечных панелей через PZEM-017 (DC)
sensor:
  - platform: pzem004t    # PZEM-017 DC вариант
    name: "Солнечная панель — мощность"
    power:
      name: "Солнце — мощность"
    energy:
      name: "Солнце — выработка"
      state_class: total_increasing
    voltage:
      name: "Солнце — напряжение"
    current:
      name: "Солнце — ток"

# HA automation: оповещение о перегреве кровли
- alias: "Чердак: перегрев"
  trigger:
    - platform: numeric_state
      entity_id: sensor.attic_temp
      above: 55
  action:
    - service: notify.mobile_app_owner
      data:
        message: "Чердак: {{ states('sensor.attic_temp') }}°C — риск перегрева кровли"

# HA automation: высокая влажность → грибок риск
- alias: "Чердак: высокая влажность"
  trigger:
    - platform: numeric_state
      entity_id: sensor.attic_humidity
      above: 80
  action:
    - service: notify.mobile_app_owner
      data:
        message: "Чердак: влажность {{ states('sensor.attic_humidity') }}% — риск конденсата"

Мансардные окна (автоматика)

# ESP32: автоматическое закрытие при дожде / граде
binary_sensor:
  - platform: homeassistant
    name: "Погода: идёт дождь"
    entity_id: binary_sensor.rain_sensor

cover:
  - platform: gpio
    name: "Мансардное окно"
    open_action:
      - switch.turn_on: servo_open
      - delay: 10s
      - switch.turn_off: servo_open
    close_action:
      - switch.turn_on: servo_close
      - delay: 10s
      - switch.turn_off: servo_close

# HA automation
- alias: "Мансарда: закрыть окно при дожде"
  trigger:
    - platform: state
      entity_id: binary_sensor.rain_sensor
      to: "on"
  action:
    - service: cover.close_cover
      target:
        entity_id: cover.mansarda_window
