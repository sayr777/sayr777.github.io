# ESPHome: котельная — датчики и реле
sensor:
  # Температура котла (DS18B20)
  - platform: dallas_temp
    address: 0x1c3c01d6076bf128
    name: "Котёл — температура подачи"

  - platform: dallas_temp
    address: 0xf83c01d607923c28
    name: "Котёл — температура обратки"

  # Давление в системе
  - platform: adc
    pin: GPIO34
    name: "Давление системы"
    filters:
      - lambda: return (x * 6.0);   # 0–3.3В → 0–6 бар
    unit_of_measurement: "bar"

  # ΔT (КПД котла)
  - platform: template
    name: "Котёл ΔT"
    lambda: >
      return id(boiler_supply).state - id(boiler_return).state;
    unit_of_measurement: "°C"

switch:
  - platform: gpio
    name: "Циркуляционный насос"
    pin: GPIO26

  - platform: gpio
    name: "Зона 1 — гостиная"
    pin: GPIO27

  - platform: gpio
    name: "Зона 2 — спальня"
    pin: GPIO14

# Подвал — вводной щит
Компоненты:
  SDM630 (трёхфазный вводной):
    → sensor.energy_main (HA Energy Dashboard)
  PZEM-004T × 4 (субсчётчики):
    → sensor.energy_house
    → sensor.energy_garage
    → sensor.energy_guest
    → sensor.energy_sauna
  UPS / АКБ (12В 100 Ач):
    → PZEM-017 DC мониторинг
    → Зарядка от солнечных панелей (MPPT)

Сенсоры безопасности

# Подвал требует отдельного мониторинга
binary_sensor:
  # Датчик затопления (утечка воды)
  - platform: gpio
    name: "Подвал: затопление"
    device_class: moisture
    pin:
      number: GPIO35
      mode: INPUT_PULLUP

  # Датчик угарного газа (CO)
  - platform: gpio
    name: "Подвал: угарный газ (CO)"
    device_class: gas
    pin: GPIO32
    on_press:
      - homeassistant.event:
          event: esphome.basement_co_alert

  # Движение в кладовой
  - platform: gpio
    name: "Подвал: движение"
    device_class: motion
    pin: GPIO33

# HA automations
- alias: "Подвал: затопление"
  trigger:
    - platform: state
      entity_id: binary_sensor.basement_flood
      to: "on"
  action:
    - service: notify.mobile_app_owner
      data:
        title: "Затопление в подвале!"
        message: "Проверьте немедленно"
    - service: switch.turn_off
      target:
        entity_id: switch.pump_circulation   # остановить насос

- alias: "Подвал: угарный газ"
  trigger:
    - platform: event
      event_type: esphome.basement_co_alert
  action:
    - service: notify.mobile_app_owner
      data:
        title: "УГАРНЫЙ ГАЗ в подвале!"
        message: "Немедленно проветрить и покинуть помещение"
    - service: siren.turn_on
      target:
        entity_id: siren.alarm

Вентиляция подвала

# ESP32: вентилятор вытяжной (влажность / CO2)
sensor:
  - platform: bme680
    humidity:
      name: "Подвал — влажность"
      id: basement_humidity
    gas_resistance:
      name: "Подвал — воздух"

switch:
  - platform: gpio
    name: "Вентилятор подвал"
    pin: GPIO25

# HA automation: вентиляция
- alias: "Подвал: вентиляция по влажности"
  trigger:
    - platform: numeric_state
      entity_id: sensor.basement_humidity
      above: 75
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.fan_basement
