# HA automations: хозблок
- alias: "Хозблок: выключить инструмент (таймер)"
  trigger:
    - platform: state
      entity_id: switch.plug_tool_shed
      to: "on"
      for: "04:00:00"
  action:
    - service: switch.turn_off
      target:
        entity_id: switch.plug_tool_shed
    - service: notify.mobile_app_owner
      data:
        message: "Хозблок: розетка инструмента выключена (4 часа)"

- alias: "Хозблок: движение ночью"
  trigger:
    - platform: state
      entity_id: binary_sensor.motion_shed
      to: "on"
  condition:
    - condition: time
      after: "23:00:00"
      before: "06:00:00"
  action:
    - service: notify.mobile_app_owner
      data:
        title: "Движение в хозблоке!"
        message: "Ночью зафиксировано движение"
    - service: light.turn_on
      target:
        entity_id: light.outdoor_flood_light

Теплица

Теплица требует постоянного контроля: полив, вентиляция, освещение при пасмурной погоде.

# esphome/greenhouse.yaml
sensor:
  - platform: bme280_i2c
    temperature:
      name: "Теплица — температура"
      id: gh_temp
    humidity:
      name: "Теплица — влажность"
      id: gh_humidity
    address: 0x76

  # Влажность почвы (резистивный датчик)
  - platform: adc
    pin: GPIO34
    name: "Теплица — влажность почвы"
    id: soil_moisture
    filters:
      - lambda: return (1.0 - (x / 3.3)) * 100.0;   # 100% = сухо, 0% = мокро
    unit_of_measurement: "%"

switch:
  # Полив (электромагнитный клапан 12В)
  - platform: gpio
    name: "Теплица — полив"
    id: irrigation_valve
    pin: GPIO26
    restore_mode: RESTORE_DEFAULT_OFF

  # Форточка (сервопривод)
  - platform: gpio
    name: "Теплица — форточка"
    pin: GPIO27

  # Досветка (фитолампа 220В через реле)
  - platform: gpio
    name: "Теплица — фитолампа"
    pin: GPIO25

# HA automations: теплица
- alias: "Теплица: полив по расписанию"
  trigger:
    - platform: time
      at: "07:00:00"
    - platform: time
      at: "19:00:00"
  condition:
    - condition: numeric_state
      entity_id: sensor.greenhouse_soil_moisture
      above: 40   # не поливать если достаточно влажно
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.greenhouse_irrigation
    - delay: "00:02:00"
    - service: switch.turn_off
      target:
        entity_id: switch.greenhouse_irrigation

- alias: "Теплица: перегрев → открыть форточку"
  trigger:
    - platform: numeric_state
      entity_id: sensor.greenhouse_temperature
      above: 30
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.greenhouse_vent

- alias: "Теплица: холодно → закрыть форточку"
  trigger:
    - platform: numeric_state
      entity_id: sensor.greenhouse_temperature
      below: 18
  action:
    - service: switch.turn_off
      target:
        entity_id: switch.greenhouse_vent

- alias: "Теплица: досветка при пасмурной погоде"
  trigger:
    - platform: time
      at: "08:00:00"
  condition:
    - condition: numeric_state
      entity_id: sensor.outdoor_illuminance
      below: 5000   # люкс
    - condition: time
      before: "20:00:00"
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.greenhouse_phyto_lamp

Беседка / Навес

# Беседка: уличное освещение + розетки для праздников
switch:
  - name: "Беседка — гирлянда"
    platform: gpio
    pin: GPIO32

  - name: "Беседка — розетки (уличные)"
    platform: gpio
    pin: GPIO33

# HA: включить гирлянду по закату
- alias: "Беседка: гирлянда по закату"
  trigger:
    - platform: sun
      event: sunset
  condition:
    - condition: state
      entity_id: input_boolean.holiday_mode
      state: "on"
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.pergola_lights

- alias: "Беседка: гирлянда по рассвету"
  trigger:
    - platform: sun
      event: sunrise
  action:
    - service: switch.turn_off
      target:
        entity_id: switch.pergola_lights

Насосная станция / Скважина

# Мониторинг насоса водоснабжения
sensor:
  - platform: pzemac
    name: "Насос скважины — мощность"
    power:
      name: "Насос — мощность"
    energy:
      name: "Насос — потребление"
      state_class: total_increasing
    uart_id: pzem_uart

# Датчик давления в системе водоснабжения
  - platform: adc
    pin: GPIO35
    name: "Давление водоснабжения"
    filters:
      - lambda: return (x * 6.0);
    unit_of_measurement: "bar"

# HA automation: насос работает слишком долго → утечка
- alias: "Насос: работает дольше 30 мин → утечка?"
  trigger:
    - platform: state
      entity_id: binary_sensor.pump_running
      to: "on"
      for: "00:30:00"
  action:
    - service: notify.mobile_app_owner
      data:
        title: "Насос работает 30 минут!"
        message: "Возможна утечка. Проверьте трубы."
