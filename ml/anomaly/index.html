Isolation Forest — аномалии паттернов доступа

Isolation Forest хорошо работает на многомерных данных без меток.

# ml/anomaly/access_anomaly.py
import numpy as np
from sklearn.ensemble import IsolationForest
from sklearn.preprocessing import StandardScaler
import joblib

def extract_features(event: dict) -> list:
    """Признаки одного события доступа"""
    return [
        event["hour"],                    # час суток (0–23)
        event["day_of_week"],             # день недели (0–6)
        event["door_zone_risk"],          # риск зоны (0.0–1.0): сервер=1.0, лобби=0.1
        event["time_since_last_entry"],   # секунд с последнего прохода
        event["entries_today"],           # сколько раз вошёл сегодня
        event["entries_this_hour"],       # сколько раз за последний час
        int(event["is_weekend"]),         # выходной день
        int(event["is_off_hours"]),       # нерабочее время
    ]

class AccessAnomalyDetector:
    def __init__(self):
        self.model = IsolationForest(
            n_estimators=200,
            contamination=0.01,    # 1% аномалий в обучающей выборке
            random_state=42,
            n_jobs=-1
        )
        self.scaler = StandardScaler()

    def fit(self, events: list[dict]):
        X = np.array([extract_features(e) for e in events])
        X_scaled = self.scaler.fit_transform(X)
        self.model.fit(X_scaled)

    def predict(self, event: dict) -> tuple[bool, float]:
        """Возвращает (is_anomaly, anomaly_score)"""
        x = np.array([extract_features(event)])
        x_scaled = self.scaler.transform(x)
        score = self.model.score_samples(x_scaled)[0]   # более отрицательный = аномальнее
        is_anomaly = self.model.predict(x_scaled)[0] == -1
        return is_anomaly, float(score)

    def save(self, path: str):
        joblib.dump({"model": self.model, "scaler": self.scaler}, path)

    @classmethod
    def load(cls, path: str):
        detector = cls.__new__(cls)
        data = joblib.load(path)
        detector.model = data["model"]
        detector.scaler = data["scaler"]
        return detector

LSTM Autoencoder — аномалии временных рядов (энергия)

Autoencoder обучается реконструировать нормальный паттерн. Высокая ошибка реконструкции = аномалия.

# ml/anomaly/energy_anomaly.py
import numpy as np
import torch
import torch.nn as nn
from torch.utils.data import DataLoader, TensorDataset

WINDOW = 144   # 144 × 10 сек = 24 часа

class LSTMAutoencoder(nn.Module):
    def __init__(self, input_size: int = 4, hidden_size: int = 64, num_layers: int = 2):
        super().__init__()
        # Encoder
        self.encoder = nn.LSTM(input_size, hidden_size, num_layers,
                               batch_first=True, dropout=0.2)
        # Decoder
        self.decoder = nn.LSTM(hidden_size, hidden_size, num_layers,
                               batch_first=True, dropout=0.2)
        self.output_layer = nn.Linear(hidden_size, input_size)

    def forward(self, x):
        # x: (batch, WINDOW, input_size)
        _, (hidden, cell) = self.encoder(x)

        # Decoder: повторяем последний hidden state
        decoder_input = hidden[-1].unsqueeze(1).repeat(1, x.size(1), 1)
        out, _ = self.decoder(decoder_input, (hidden, cell))
        reconstruction = self.output_layer(out)
        return reconstruction

def train_energy_autoencoder(data: np.ndarray) -> tuple:
    """
    data: (N, 4) — [power_w, voltage_v, current_a, power_factor]
    Возвращает (model, threshold)
    """
    # Нормализация
    mean, std = data.mean(0), data.std(0)
    data_norm = (data - mean) / (std + 1e-8)

    # Создание скользящих окон
    windows = np.array([data_norm[i:i+WINDOW] for i in range(len(data_norm)-WINDOW)])
    dataset = TensorDataset(torch.FloatTensor(windows))
    loader = DataLoader(dataset, batch_size=32, shuffle=True)

    model = LSTMAutoencoder()
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    criterion = nn.MSELoss()

    model.train()
    for epoch in range(50):
        for (batch,) in loader:
            pred = model(batch)
            loss = criterion(pred, batch)
            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

    # Вычислить порог: 95-й перцентиль ошибки на обучающих данных
    model.eval()
    errors = []
    with torch.no_grad():
        for (batch,) in loader:
            pred = model(batch)
            err = ((pred - batch) ** 2).mean(dim=(1,2)).numpy()
            errors.extend(err)
    threshold = np.percentile(errors, 95)

    return model, threshold, mean, std

def detect_energy_anomaly(model, threshold, mean, std,
                           window: np.ndarray) -> tuple[bool, float]:
    """Проверить один временной ряд длиной WINDOW"""
    norm = (window - mean) / (std + 1e-8)
    t = torch.FloatTensor(norm).unsqueeze(0)
    with torch.no_grad():
        reconstruction = model(t)
    error = float(((reconstruction - t) ** 2).mean())
    return error > threshold, error

One-Class SVM — аномалии графа движения

Каждый сотрудник имеет типичный маршрут по зонам. Нетипичный маршрут = потенциальное нарушение.

# ml/anomaly/movement_graph.py
from sklearn.svm import OneClassSVM
from sklearn.preprocessing import LabelEncoder
import numpy as np

def encode_movement_sequence(zones: list[str], all_zones: list[str]) -> np.ndarray:
    """
    Закодировать последовательность зон как вектор transition counts.
    zones: ["lobby", "floor_1", "office_a", "floor_1", "exit"]
    → матрица переходов в виде вектора
    """
    n = len(all_zones)
    zone_idx = {z: i for i, z in enumerate(all_zones)}
    transitions = np.zeros((n, n))
    for i in range(len(zones) - 1):
        a, b = zone_idx.get(zones[i], 0), zone_idx.get(zones[i+1], 0)
        transitions[a][b] += 1
    return transitions.flatten() / (transitions.sum() + 1e-8)

class MovementAnomalyDetector:
    def __init__(self, all_zones: list[str]):
        self.all_zones = all_zones
        self.models: dict[str, OneClassSVM] = {}

    def fit_person(self, person_id: str, daily_sequences: list[list[str]]):
        """Обучить на исторических маршрутах одного сотрудника"""
        X = np.array([encode_movement_sequence(seq, self.all_zones)
                      for seq in daily_sequences])
        svm = OneClassSVM(kernel="rbf", nu=0.05, gamma="scale")
        svm.fit(X)
        self.models[person_id] = svm

    def is_anomalous(self, person_id: str,
                      today_sequence: list[str]) -> tuple[bool, float]:
        if person_id not in self.models:
            return False, 0.0   # нет модели — не знаем
        svm = self.models[person_id]
        x = encode_movement_sequence(today_sequence, self.all_zones).reshape(1, -1)
        score = svm.score_samples(x)[0]
        return score < 0, score   # отрицательный = аномалия

XGBoost — Fraud Detection (EventPass)

Детектировать аномальное сканирование билетов: необычно быстро, с разных устройств, геоаномалии.

# ml/anomaly/fraud_detector.py
import xgboost as xgb
import numpy as np
import pandas as pd

FEATURES = [
    "scans_last_1min",        # сканирований за 1 мин с этого сканера
    "scans_last_10min",       # за 10 мин
    "jti_reuse_count",        # сколько раз этот jti уже видели
    "time_since_issue",       # секунд от выпуска до сканирования
    "scanner_zone_match",     # зона сканера совпадает с зоной в билете
    "event_started",          # мероприятие уже началось (0/1)
    "time_to_event_start",    # секунд до начала
    "consecutive_deny_count", # сколько deny подряд на этом сканере
    "unique_scanners_today",  # с сколькими разными сканерами уже встречался этот jti
    "batch_scan_flag",        # много сканирований подряд < 100мс (автоматический fraud)
]

def train_fraud_model(df: pd.DataFrame, labels: pd.Series) -> xgb.XGBClassifier:
    """
    df: DataFrame с колонками FEATURES
    labels: 1 = fraud, 0 = normal
    """
    model = xgb.XGBClassifier(
        n_estimators=300,
        max_depth=6,
        learning_rate=0.05,
        subsample=0.8,
        colsample_bytree=0.8,
        scale_pos_weight=99,   # сильный дисбаланс: 1% fraud
        eval_metric="auc",
        use_label_encoder=False,
        tree_method="hist"
    )
    model.fit(df[FEATURES], labels,
              eval_set=[(df[FEATURES], labels)],
              verbose=False)
    return model

def predict_fraud(model: xgb.XGBClassifier,
                   scan_features: dict) -> tuple[bool, float]:
    x = pd.DataFrame([scan_features])[FEATURES]
    prob = model.predict_proba(x)[0][1]
    return prob > 0.7, prob   # порог 0.7

CUSUM — обнаружение постепенного дрейфа (котёл, трубы)

# ml/anomaly/cusum.py
# Обнаружить постепенный рост температуры трубы (засор, неисправность)

def cusum_detector(values: list[float], target: float,
                   k: float = 0.5, h: float = 5.0) -> list[bool]:
    """
    CUSUM (Cumulative Sum) алгоритм.
    target: ожидаемое среднее
    k: slack (половина допустимого отклонения)
    h: порог тревоги
    """
    s_pos, s_neg = 0.0, 0.0
    alarms = []
    for x in values:
        s_pos = max(0, s_pos + (x - target) - k)
        s_neg = max(0, s_neg - (x - target) - k)
        alarms.append(s_pos > h or s_neg > h)
    return alarms

# Применение: трубы в подвале
# target = 20.0 (нормальная темп трубы в подвале)
# При постепенном росте → alarm до того как труба лопнет
