User Behavioral Profiling

# ml/behavioral/user_profile.py
from dataclasses import dataclass, field
from collections import defaultdict
import numpy as np

@dataclass
class UserBehaviorProfile:
    person_id: str
    # Типичное время первого входа (час) — распределение
    usual_arrival_hours: list[float] = field(default_factory=list)
    # Типичное время последнего выхода (час)
    usual_departure_hours: list[float] = field(default_factory=list)
    # Зоны которые посещает регулярно
    frequent_zones: set[str] = field(default_factory=set)
    # Зоны которые никогда не посещал
    never_visited_zones: set[str] = field(default_factory=set)
    # Среднее количество проходов в день
    avg_daily_entries: float = 0.0
    # Стандартное отклонение
    std_daily_entries: float = 0.0
    # Паттерн выходных (работает в выходные?)
    weekend_entries_avg: float = 0.0
    # Коллеги: с кем обычно приходит/уходит (±5 мин)
    usual_companions: list[str] = field(default_factory=list)

def compute_risk_score(event: dict, profile: UserBehaviorProfile) -> float:
    """
    Вычислить риск-скор события (0.0 = норма, 1.0 = максимальная тревога).
    Комбинация нескольких сигналов.
    """
    scores = []

    # 1. Необычное время
    hour = event["hour"]
    if profile.usual_arrival_hours:
        mean_arr = np.mean(profile.usual_arrival_hours)
        std_arr = np.std(profile.usual_arrival_hours) + 0.5
        time_z = abs(hour - mean_arr) / std_arr
        scores.append(min(time_z / 3.0, 1.0))   # нормализовать [0, 1]

    # 2. Нетипичная зона
    zone = event["zone_id"]
    if zone in profile.never_visited_zones:
        scores.append(0.9)
    elif zone not in profile.frequent_zones:
        scores.append(0.4)
    else:
        scores.append(0.0)

    # 3. Выходной день (если обычно не работает)
    if event["is_weekend"] and profile.weekend_entries_avg < 0.1:
        scores.append(0.7)

    # 4. Слишком много входов за день
    daily_entries = event["entries_today"]
    if profile.std_daily_entries > 0:
        daily_z = (daily_entries - profile.avg_daily_entries) / profile.std_daily_entries
        scores.append(min(max(daily_z / 3.0, 0), 1.0))

    # 5. Ночное время (00:00 – 05:00)
    if 0 <= hour <= 5:
        scores.append(0.8)

    return float(np.max(scores)) if scores else 0.0

Graph-Based Movement Analysis

Строим граф переходов между зонами. Нетипичный путь = высокий риск.

# ml/behavioral/movement_graph.py
import networkx as nx
from collections import defaultdict

class MovementGraphAnalyzer:
    """
    Для каждого сотрудника строит ориентированный граф его движений.
    Рёбра взвешены: вес = частота перехода.
    """

    def build_personal_graph(self, events: list[dict]) -> nx.DiGraph:
        G = nx.DiGraph()
        transitions = defaultdict(int)

        for i in range(len(events) - 1):
            if events[i]["person_id"] == events[i+1]["person_id"]:
                src = events[i]["zone_id"]
                dst = events[i+1]["zone_id"]
                transitions[(src, dst)] += 1

        total = sum(transitions.values())
        for (src, dst), count in transitions.items():
            G.add_edge(src, dst, weight=count/total, count=count)

        return G

    def path_probability(self, G: nx.DiGraph, path: list[str]) -> float:
        """Вероятность наблюдаемого пути для данного сотрудника"""
        if len(path) < 2:
            return 1.0

        prob = 1.0
        for i in range(len(path) - 1):
            src, dst = path[i], path[i+1]
            if G.has_edge(src, dst):
                prob *= G[src][dst]["weight"]
            else:
                prob *= 0.001   # редкий переход = маловероятен
        return prob

    def detect_unusual_path(self, G: nx.DiGraph, path: list[str],
                             threshold: float = 0.01) -> bool:
        prob = self.path_probability(G, path)
        return prob < threshold

BLE RSSI Fingerprinting (DN·Beacon)

Определить точное местоположение человека по отпечатку RSSI от нескольких считывателей.

# ml/behavioral/ble_localization.py
# K-NN indoor localization по RSSI fingerprint
# Каждая точка пространства имеет уникальный вектор RSSI от N считывателей

import numpy as np
from sklearn.neighbors import KNeighborsClassifier

class BLELocationModel:
    def __init__(self, n_neighbors: int = 5):
        self.model = KNeighborsClassifier(n_neighbors=n_neighbors,
                                          metric="euclidean",
                                          weights="distance")
        self.scanner_ids: list[str] = []
        self.zone_labels: list[str] = []

    def collect_fingerprint(self, zone: str,
                              rssi_by_scanner: dict[str, float]) -> dict:
        """Запись обучающей точки. Нужно ~20 замеров на зону."""
        return {"zone": zone, "rssi": rssi_by_scanner}

    def fit(self, fingerprints: list[dict]):
        self.scanner_ids = list(set(
            k for fp in fingerprints for k in fp["rssi"]
        ))
        X = []
        y = []
        for fp in fingerprints:
            vector = [fp["rssi"].get(sid, -100) for sid in self.scanner_ids]
            X.append(vector)
            y.append(fp["zone"])
        self.model.fit(X, y)
        self.zone_labels = list(set(y))

    def predict_zone(self, rssi_by_scanner: dict[str, float]) -> tuple[str, float]:
        vector = [rssi_by_scanner.get(sid, -100) for sid in self.scanner_ids]
        zone = self.model.predict([vector])[0]
        probs = self.model.predict_proba([vector])[0]
        confidence = float(max(probs))
        return zone, confidence

# Применение:
# DN·Beacon браслет → 3–5 ESP32 считывателей → RSSI вектор →
# BLELocationModel.predict_zone() → комната с точностью ±1м

Continuous Authentication (поведенческая биометрия)

Не только “кто вошёл”, но и “всё ещё ли это тот человек после входа”.

# ml/behavioral/continuous_auth.py
# Анализ паттерна движения по BLE RSSI во времени
# "Цифровая подпись" перемещений уникальна для каждого человека

from sklearn.preprocessing import StandardScaler
from sklearn.svm import SVC
import numpy as np

class ContinuousAuthenticator:
    """
    Обучается на характерных паттернах движения конкретного человека.
    Если наблюдаемый паттерн не совпадает → возможная подмена.
    Применение: важные зоны, серверные, кассы.
    """

    def __init__(self):
        self.classifiers: dict[str, SVC] = {}
        self.scalers: dict[str, StandardScaler] = {}

    def extract_movement_features(self,
                                   rssi_timeseries: list[dict]) -> np.ndarray:
        """
        rssi_timeseries: последние 30 секунд RSSI замеров
        Признаки: скорость изменения RSSI, дисперсия, переходы между комнатами
        """
        features = []
        for scanner_id in self.scanner_ids:
            values = [r.get(scanner_id, -100) for r in rssi_timeseries]
            features += [
                np.mean(values),
                np.std(values),
                np.max(values) - np.min(values),     # range
                np.mean(np.diff(values)),             # mean velocity
                np.std(np.diff(values)),              # velocity variance
            ]
        return np.array(features)

    def fit_person(self, person_id: str,
                    sessions: list[list[dict]]):
        X = np.array([self.extract_movement_features(s) for s in sessions])
        # Positive class: этот человек
        # Negative: случайный шум (генерируем синтетически)
        noise = np.random.uniform(-100, -50, X.shape)
        X_train = np.vstack([X, noise])
        y_train = np.array([1]*len(X) + [0]*len(noise))

        scaler = StandardScaler()
        X_scaled = scaler.fit_transform(X_train)

        svm = SVC(kernel="rbf", C=1.0, probability=True)
        svm.fit(X_scaled, y_train)

        self.classifiers[person_id] = svm
        self.scalers[person_id] = scaler

    def verify(self, person_id: str,
                recent_rssi: list[dict]) -> tuple[bool, float]:
        if person_id not in self.classifiers:
            return True, 1.0   # нет профиля — не отклонять

        features = self.extract_movement_features(recent_rssi).reshape(1, -1)
        scaled = self.scalers[person_id].transform(features)
        prob = self.classifiers[person_id].predict_proba(scaled)[0][1]
        return prob > 0.6, prob

            "path": round(path_risk, 3),
            "location": round(location_risk, 3),
        },
        "recommendation": (
            "deny" if final_score > 0.8 else
            "require_2fa" if final_score > 0.5 else
            "allow"
        )

Интеграция с Keycloak PEP

# tenantpass/pep_agent.py — ML risk score в решении доступа

async def check_access(person_id: str, door_id: str, event: dict) -> bool:
    # Сначала: базовая проверка прав (Keycloak/кэш)
    basic_permit = await check_basic_permit(person_id, door_id)
    if not basic_permit:
        return False

    # Затем: ML risk score
    async with httpx.AsyncClient() as c:
        r = await c.post("http://ml-risk-api:8000/risk_score", json=event)
    risk = r.json()

    if risk["recommendation"] == "deny":
        log.warning("ML deny: %s at %s, score=%.2f", person_id, door_id, risk["risk_score"])
        await alert_security(person_id, door_id, risk)
        return False
    elif risk["recommendation"] == "require_2fa":
        # Запросить дополнительное подтверждение (PIN / биометрия)
        return await request_2fa(person_id, door_id)
    else:
        return True
