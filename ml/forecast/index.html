Prophet — прогноз потока на КПП

# ml/forecast/kpp_flow.py
# Facebook Prophet обрабатывает сезонности (день/неделя/год) автоматически

from prophet import Prophet
import pandas as pd

def train_kpp_forecast(events: pd.DataFrame) -> Prophet:
    """
    events: DataFrame с колонками ds (datetime), y (count)
    y = количество проходов в 15-минутные интервалы
    """
    # Агрегация по 15-минутным окнам
    df = events.copy()
    df["ds"] = df["occurred_at"].dt.floor("15min")
    agg = df.groupby("ds").size().reset_index(name="y")

    model = Prophet(
        yearly_seasonality=True,
        weekly_seasonality=True,
        daily_seasonality=True,
        changepoint_prior_scale=0.1,      # гибкость тренда
        seasonality_prior_scale=10.0,
    )

    # Добавить праздники РФ
    model.add_country_holidays(country_name="RU")

    # Регрессоры: температура, день зарплаты
    model.add_regressor("outdoor_temp")
    model.add_regressor("is_payday")

    model.fit(agg)
    return model

def predict_next_4h(model: Prophet,
                    current_temp: float,
                    is_payday: bool) -> pd.DataFrame:
    future = model.make_future_dataframe(periods=16, freq="15min")
    future["outdoor_temp"] = current_temp
    future["is_payday"] = int(is_payday)
    forecast = model.predict(future)
    return forecast["ds", "yhat", "yhat_lower", "yhat_upper"](/"ds", "yhat", "yhat_lower", "yhat_upper"/).tail(16)

# Применение:
# Если forecast.yhat > N (текущая пропускная способность) →
# → открыть дополнительный турникет за 30 мин до пика

LightGBM — прогноз возврата домой (HomePass)

# ml/forecast/homecoming.py
# Предсказать когда хозяин вернётся домой
# Input: BLE-маяк пропал из дома, GPS последнее положение, время суток

import lightgbm as lgb
import pandas as pd
import numpy as np

FEATURES_HOMECOMING = [
    "departure_hour",        # час ухода
    "departure_dow",         # день недели ухода
    "usual_return_hour",     # обычный час возврата (из истории)
    "is_workday",            # рабочий день
    "weather_icon",          # погода (encoded)
    "last_gps_zone",         # откуда ушёл (работа/магазин/кино)
    "phone_battery",         # заряд телефона (если > 80% — вряд ли торопится)
    "calendar_event_in_2h",  # в Google Calendar есть событие через 2ч (0/1)
    "days_since_last_late",  # сколько дней прошло с последнего позднего прихода
]

def train_homecoming_model(history: pd.DataFrame) -> lgb.LGBMRegressor:
    """
    history: история возвращений
    target: минут до возвращения с момента ухода
    """
    model = lgb.LGBMRegressor(
        n_estimators=500,
        learning_rate=0.05,
        num_leaves=31,
        feature_fraction=0.8,
        bagging_fraction=0.8,
        bagging_freq=5,
    )
    model.fit(
        history[FEATURES_HOMECOMING],
        history["minutes_until_return"],
        eval_set=[(history[FEATURES_HOMECOMING], history["minutes_until_return"])]
    )
    return model

async def predict_homecoming(model, current_features: dict) -> int:
    """Вернуть предсказанное количество минут до прихода"""
    x = pd.DataFrame([current_features])[FEATURES_HOMECOMING]
    minutes = int(model.predict(x)[0])
    return max(5, minutes)   # минимум 5 минут

# HA automation:
# Если predicted_return_in_minutes < 30 → включить котёл, свет, вентиляцию

LSTM — прогноз энергопотребления

# ml/forecast/energy_forecast.py
# Прогноз потребления на 24ч → оптимальное включение устройств по тарифу

import torch
import torch.nn as nn
import numpy as np

class EnergyForecastLSTM(nn.Module):
    def __init__(self, input_size: int = 6, hidden: int = 128,
                 horizon: int = 96):   # 96 × 15мин = 24ч
        super().__init__()
        self.lstm = nn.LSTM(input_size, hidden, num_layers=2,
                            batch_first=True, dropout=0.2)
        self.head = nn.Sequential(
            nn.Linear(hidden, 64),
            nn.ReLU(),
            nn.Linear(64, horizon)
        )

    def forward(self, x):
        out, _ = self.lstm(x)
        return self.head(out[:, -1, :])

# Признаки:
# [power_w, hour_sin, hour_cos, dow_sin, dow_cos, outdoor_temp]
# Циклические признаки: sin/cos кодирование для часа и дня недели

def encode_cyclical(value: float, period: float) -> tuple[float, float]:
    return (np.sin(2 * np.pi * value / period),
            np.cos(2 * np.pi * value / period))

# Применение: ночной тариф оптимизация
# Если forecast[22:00] > threshold → включить бойлер в 21:50 (пока тариф ниже)
# Если forecast[07:00] peak → включить стирку в 05:00

Predictive Maintenance — Survival Analysis

# ml/forecast/maintenance.py
# Предсказать когда оборудование выйдет из строя
# Алгоритм: Cox Proportional Hazards (lifelines)

from lifelines import CoxPHFitter
import pandas as pd

def train_maintenance_model(equipment_history: pd.DataFrame) -> CoxPHFitter:
    """
    equipment_history:
      - duration: дней до отказа (или цензурирования)
      - event: 1=отказ, 0=ещё работает (цензурировано)
      - operating_hours: часов наработки
      - avg_temp_c: средняя температура работы
      - voltage_variance: дисперсия напряжения
      - restart_count_30d: перезапусков за 30 дней
    """
    cph = CoxPHFitter(penalizer=0.1)
    cph.fit(
        equipment_history,
        duration_col="duration",
        event_col="event"
    )
    return cph

def days_until_failure(model: CoxPHFitter,
                        device_features: dict,
                        survival_probability: float = 0.3) -> int:
    """
    Вернуть количество дней до момента когда
    вероятность выживания упадёт ниже порога (default 30%)
    """
    df = pd.DataFrame([device_features])
    survival_fn = model.predict_survival_function(df)
    # Найти t где S(t) = survival_probability
    times = survival_fn.index.values
    probs = survival_fn.iloc[:, 0].values
    for t, p in zip(times, probs):
        if p < survival_probability:
            return int(t)
    return 365   # больше года — обслуживание не срочно

# Применение:
# Каждую ночь → вычислить days_until_failure для каждого устройства
# Если < 30 дней → создать задачу обслуживания в HA
# Если < 7 дней → уведомить немедленно

SARIMA — очереди на мероприятии (EventPass)

# ml/forecast/queue_forecast.py
# Сезонный ARIMA для прогноза длины очереди на вход

from statsmodels.tsa.statespace.sarimax import SARIMAX
import pandas as pd

def train_queue_model(queue_lengths: pd.Series) -> SARIMAX:
    """
    queue_lengths: временной ряд (15-сек интервалы)
    длина очереди у входа на стадион/концерт
    """
    model = SARIMAX(
        queue_lengths,
        order=(2, 1, 2),           # ARIMA(p,d,q)
        seasonal_order=(1, 0, 1, 4),  # сезонность 1 мин (4 × 15 сек)
        enforce_stationarity=False,
        enforce_invertibility=False
    )
    result = model.fit(disp=False)
    return result

def predict_queue_2min(result, steps: int = 8) -> pd.Series:
    """Прогноз на 2 минуты (8 × 15 сек)"""
    return result.forecast(steps=steps)

# Применение:
# Если forecast > capacity → за 2 мин до пика открыть дополнительный вход
# Уведомить контролёра о предстоящей перегрузке
