Feature Store (Redis Streams)

Real-time признаки для онлайн-inference.

# ml/features/feature_store.py
import redis
import json
from datetime import datetime

class FeatureStore:
    def __init__(self, redis_url: str):
        self.r = redis.from_url(redis_url)

    def update_person_features(self, person_id: str, features: dict):
        """Обновить признаки пользователя (вызывается при каждом событии)"""
        key = f"features:person:{person_id}"
        self.r.hset(key, mapping={
            k: json.dumps(v) for k, v in features.items()
        })
        self.r.expire(key, 86400)   # 24ч TTL

    def get_person_features(self, person_id: str) -> dict:
        key = f"features:person:{person_id}"
        raw = self.r.hgetall(key)
        return {k.decode(): json.loads(v) for k, v in raw.items()}

    def push_event(self, stream: str, event: dict):
        """Добавить событие в Redis Stream для real-time обработки"""
        self.r.xadd(stream, event, maxlen=10000)

    def get_recent_events(self, person_id: str,
                           minutes: int = 60) -> list[dict]:
        """События пользователя за последние N минут"""
        start_ms = int((datetime.now().timestamp() - minutes * 60) * 1000)
        events = self.r.xrange(f"events:{person_id}",
                               min=f"{start_ms}-0", max="+")
        return [e[1] for e in events]

# Признаки которые обновляются в реальном времени:
# person:entries_today        → инкремент при каждом входе
# person:last_seen_zone       → обновляется каждый проход
# person:rssi_snapshot        → обновляется каждые 30 сек от BLE
# person:current_risk_score   → пересчитывается каждую минуту

Мониторинг ML-моделей (Data Drift)

# ml/monitoring/drift_detector.py
# Обнаружить дрейф данных: модель устарела, нужно переобучение

from scipy.stats import ks_2samp
import numpy as np

def detect_feature_drift(reference: np.ndarray,
                          current: np.ndarray,
                          threshold: float = 0.05) -> dict:
    """
    Kolmogorov-Smirnov тест для каждого признака.
    p-value < threshold → дрейф обнаружен → нужно переобучение.
    """
    results = {}
    for i in range(reference.shape[1]):
        stat, p_value = ks_2samp(reference[:, i], current[:, i])
        results[f"feature_{i}"] = {
            "ks_statistic": float(stat),
            "p_value": float(p_value),
            "drift_detected": p_value < threshold
        }
    return results

def check_prediction_drift(model_predictions_last_week: list[float],
                            model_predictions_this_week: list[float]) -> bool:
    """PSI (Population Stability Index) для мониторинга предсказаний"""
    # PSI > 0.2 → значительный дрейф → переобучить
    hist_ref, bins = np.histogram(model_predictions_last_week, bins=10)
    hist_cur, _ = np.histogram(model_predictions_this_week, bins=bins)
    hist_ref = (hist_ref + 1e-8) / len(model_predictions_last_week)
    hist_cur = (hist_cur + 1e-8) / len(model_predictions_this_week)
    psi = np.sum((hist_ref - hist_cur) * np.log(hist_ref / hist_cur))
    return psi > 0.2

# Prefect: запускать мониторинг ежедневно
# При drift_detected → запустить переобучение → MLflow → Triton

# Запуск ONNX модели прямо на RPi (без Triton)
# Для: offline EventPass сканер, HomePass локальный inference

import onnxruntime as ort
import numpy as np

session = ort.InferenceSession(
    "models/access_anomaly.onnx",
    providers=["CPUExecutionProvider"]
)

def predict_anomaly_edge(features: np.ndarray) -> float:
    input_name = session.get_inputs()[0].name
    result = session.run(None, {input_name: features.reshape(1, -1).astype(np.float32)})
    return float(result[0][0])   # anomaly score
