Face Recognition — CompreFace

sequenceDiagram participant F as Frigate participant DT as Double Take participant CF as CompreFace participant KC as Keycloak / HA F->>DT: person detected (snapshot) DT->>CF: POST /api/v1/recognition/recognize Note over CF: ArcFace embedding → cosine similarity CF-->>DT: [{subject: "Иванов", confidence: 0.97}] DT->>KC: user_recognized("Иванов", door_id, confidence) KC->>KC: если confidence > 0.85 AND карта → permit
# ml/face/compreface_client.py
import httpx
from pathlib import Path

CF_URL = "http://compreface:8000"
RECOGNITION_KEY = "your-api-key"

async def recognize_face(image_path: str) -> dict:
    async with httpx.AsyncClient() as c:
        with open(image_path, "rb") as f:
            r = await c.post(
                f"{CF_URL}/api/v1/recognition/recognize",
                headers={"x-api-key": RECOGNITION_KEY},
                files={"file": f},
                params={"face_plugins": "age,gender,landmarks",
                        "limit": 1,
                        "prediction_count": 3,
                        "det_prob_threshold": 0.8}
            )
    result = r.json()
    if result.get("result"):
        best = result["result"][0]["subjects"][0]
        return {
            "subject": best["subject"],
            "confidence": best["similarity"],
            "age": result["result"][0].get("age"),
        }
    return {"subject": "unknown", "confidence": 0.0}

async def enroll_face(person_id: str, image_path: str):
    """Добавить лицо сотрудника в базу"""
    async with httpx.AsyncClient() as c:
        with open(image_path, "rb") as f:
            await c.post(
                f"{CF_URL}/api/v1/recognition/faces",
                headers={"x-api-key": RECOGNITION_KEY},
                files={"file": f},
                params={"subject": person_id}
            )

License Plate Recognition

# ml/lpr/plate_reader.py
# Используем PaddleOCR (Apache 2.0) для распознавания русских номеров

from paddleocr import PaddleOCR
import cv2, re

ocr = PaddleOCR(use_angle_cls=True, lang="en",
                det_model_dir="models/paddle/det",
                rec_model_dir="models/paddle/rec_cyrillic")

PLATE_PATTERN = re.compile(r"[АВЕКМНОРСТУХ]\d{3}[АВЕКМНОРСТУХ]{2}\d{2,3}", re.IGNORECASE)

def recognize_plate(image: "np.ndarray") -> str | None:
    results = ocr.ocr(image, cls=True)
    for line in (results[0] or []):
        text = line[1][0].upper().replace(" ", "")
        if PLATE_PATTERN.match(text):
            return text
    return None

Crowd Density Estimation

Используется в EventPass: сколько людей в зоне без подсчёта каждого бокса.

# ml/crowd/density_model.py
# CSRNet — предсказывает карту плотности → сумма = количество людей
# Работает при перекрытиях и большой плотности

import torch
from torchvision import transforms

class CrowdCounter:
    def __init__(self, model_path: str):
        self.model = torch.load(model_path, map_location="cpu")
        self.model.eval()
        self.transform = transforms.Compose([
            transforms.ToTensor(),
            transforms.Normalize([0.485,0.456,0.406],[0.229,0.224,0.225])
        ])

    def count(self, image: "PIL.Image") -> float:
        with torch.no_grad():
            tensor = self.transform(image).unsqueeze(0)
            density_map = self.model(tensor)
            return float(density_map.sum())   # количество людей

# Интеграция с EventPass
# Каждые 30 сек → снэпшот зоны → CrowdCounter → MQTT → HA
# Если count > zone.capacity → тревога организатору

Tailgating Detection (хвостовой заход)

Два человека проходят через одну точку доступа за одну авторизацию.

# ml/access/tailgating_detector.py
# Метод: подсчёт людей в bbox зоны турникета до и после авторизации

from collections import deque
from dataclasses import dataclass
from datetime import datetime

@dataclass
class TailgatingEvent:
    door_id: str
    authorized_person: str
    extra_persons_count: int
    timestamp: datetime
    confidence: float

class TailgatingDetector:
    def __init__(self):
        self.auth_events = deque(maxlen=100)   # последние авторизации

    def on_access_granted(self, person_id: str, door_id: str,
                          person_count_in_zone: int) -> TailgatingEvent | None:
        """
        Вызывается после успешной авторизации.
        person_count_in_zone — количество людей в зоне турникета
        (из YOLO детекции за последние 2 секунды)
        """
        if person_count_in_zone > 1:
            extra = person_count_in_zone - 1
            confidence = min(0.5 + extra * 0.3, 0.99)
            return TailgatingEvent(
                door_id=door_id,
                authorized_person=person_id,
                extra_persons_count=extra,
                timestamp=datetime.utcnow(),
                confidence=confidence
            )
        return None

Anti-Spoofing (Liveness Detection)

Защита от атаки фотографией при биометрии. Особенно важно для EventPass.

# ml/face/liveness.py
# Модель: MN3 или Silent-Face Anti-Spoofing (PyTorch)
# Обучена различать живое лицо и фотографию/экран

import torch
import torchvision.transforms as T

class LivenessDetector:
    def __init__(self):
        self.model = torch.jit.load("models/silent_face.pt")
        self.model.eval()
        self.transform = T.Compose([
            T.Resize((256, 256)),
            T.ToTensor(),
            T.Normalize([0.5,0.5,0.5],[0.5,0.5,0.5])
        ])

    def is_live(self, face_crop: "PIL.Image", threshold: float = 0.7) -> tuple[bool, float]:
        with torch.no_grad():
            t = self.transform(face_crop).unsqueeze(0)
            score = self.model(t).sigmoid().item()
        return score > threshold, score

# Использование при биометрическом входе:
# 1. Frigate → detect face → crop
# 2. LivenessDetector.is_live() → score
# 3. Если score < 0.7 → отказать + зафиксировать попытку спуфинга

Vision LLM (LLaVA через Ollama)

# ml/vision_llm/scene_analyzer.py
# Описывает что происходит на камере → человеко-читаемый отчёт + алерты

import httpx, base64

async def analyze_scene(image_path: str, context: str = "") -> str:
    with open(image_path, "rb") as f:
        image_b64 = base64.b64encode(f.read()).decode()

    prompt = f"""Ты система безопасности. Опиши что ты видишь на изображении.
Контекст: {context}
Обрати внимание на:
- Количество людей и их действия
- Подозрительное поведение
- Наличие или отсутствие защитных средств (каска, жилет)
- Любые аномалии

Ответ в формате JSON: {{
  "description": "...",
  "person_count": N,
  "anomaly_detected": true/false,
  "anomaly_description": "...",
  "ppe_violations": []
}}"""

    async with httpx.AsyncClient(timeout=30) as c:
        r = await c.post("http://ollama:11434/api/generate", json={
            "model": "llava:7b",
            "prompt": prompt,
            "images": [image_b64],
            "stream": False,
            "format": "json"
        })
    return r.json()["response"]

# - Пожар / дым → немедленный алерт
# - Человек упал → man-down алерт
# - Нарушение ППЭ на стройплощадке → уведомление прораба
# - Подозрительное поведение у банкомата/сейфа → тревога охране
