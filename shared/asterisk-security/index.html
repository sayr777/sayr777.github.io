Безопасность Asterisk: изоляция контекстов (замечание 12)

Безопасность Asterisk: изоляция контекстов (замечание 12)

Разделение PA / телефонии / скрытых микрофонов

Проблема

Один Asterisk для трёх use-case:

  • PA-оповещения (публичные объявления)
  • Телефония (внутренняя АТС)
  • Скрытые микрофоны (ChanSpy/MixMonitor — прослушивание зон)

Если злоумышленник получает SIP-учётку PA-спикера → может через ChanSpy прослушать любой канал.

Решение: изолированные контексты + отдельные credentials

extensions.conf — строгая изоляция

; /etc/asterisk/extensions.conf

; === PA КОНТЕКСТ ===
; PA-колонки могут только принять вызов, не могут звонить наружу
[pa-only]
; Входящий вызов от AMI → проиграть аудио
exten => _pa-*,1,Answer()
exten => _pa-*,n,Playback(${ARG1})
exten => _pa-*,n,Hangup()

; PA-колонка не может инициировать исходящий → контекст не содержит таких правил
; Попытка позвонить → invalid extension → Hangup

; === ТЕЛЕФОНИЯ КОНТЕКСТ ===
[internal-phones]
; Внутренние звонки
exten => _1XX,1,Dial(PJSIP/${EXTEN},30)
exten => _1XX,n,Hangup()

; Домофон → звонок охраннику / владельцу
exten => door-entrance,1,Dial(PJSIP/guard&PJSIP/mobile,30)
exten => door-entrance,n,Hangup()

; Явно запрещаем ChanSpy из этого контекста
; (нет extension → Asterisk не найдёт и отклонит)

; === КОНТЕКСТ МОНИТОРИНГА ЗОН ===
[zone-monitors]
; Микрофон зоны регистрируется здесь и немедленно начинает запись
exten => _mic-*,1,Answer()
exten => _mic-*,n,Set(ZONE_ID=${EXTEN})
exten => _mic-*,n,MixMonitor(/var/spool/asterisk/monitor/${ZONE_ID}-${STRFTIME(%Y%m%d_%H%M%S,UTC,)}.wav)
exten => _mic-*,n,Wait(86400)  ; ждём 24ч (автоматическая ротация)
exten => _mic-*,n,Hangup()

; ChanSpy только через явный protected extension
; Доступен ТОЛЬКО из AMI ami-monitor-user (не из SIP)
exten => spy-authorized,1,ChanSpy(PJSIP/mic-${ARG1},qE(beep))
exten => spy-authorized,n,Hangup()

pjsip.conf — разные пароли и контексты

; /etc/asterisk/pjsip.conf

; === PA КОЛОНКИ ===
[pa-zone-1]
type=endpoint
context=pa-only          ; ← изолированный контекст
auth=pa-zone-1-auth
aors=pa-zone-1-aor
disallow=all
allow=ulaw,alaw

[pa-zone-1-auth]
type=auth
auth_type=userpass
username=pa-zone-1
password=Xk9mP2qL8nRv    ; уникальный длинный пароль

; === ТЕЛЕФОНЫ ===
[phone-101]
type=endpoint
context=internal-phones   ; ← отдельный контекст
auth=phone-101-auth
aors=phone-101-aor

; === МИКРОФОНЫ ЗОН ===
[mic-zone-lobby]
type=endpoint
context=zone-monitors    ; ← отдельный контекст
auth=mic-zone-lobby-auth
aors=mic-zone-lobby-aor
; ДОПОЛНИТЕЛЬНО: только с конкретного IP (сам микрофон)

[mic-zone-lobby-auth]
type=auth
auth_type=userpass
username=mic-zone-lobby
password=Yt7vQ3sN6wXm    ; отдельный пароль, не совпадает с PA

manager.conf — ограниченные AMI-учётки

; /etc/asterisk/manager.conf

[general]
enabled=yes
bindaddr=127.0.0.1      ; только localhost! Никакого 0.0.0.0

; === AMI для PA-системы (HA/автоматизация) ===
[ami-pa-user]
secret=pa-ami-secret-long
deny=0.0.0.0/0.0.0.0
permit=127.0.0.1/255.255.255.255
permit=192.168.1.100/255.255.255.255  ; только HA-сервер
read=system,call
write=originate           ; ТОЛЬКО Originate — ничего другого

; === AMI для мониторинга ===
[ami-monitor-user]
secret=monitor-ami-secret-long
deny=0.0.0.0/0.0.0.0
permit=127.0.0.1/255.255.255.255
read=system,call
write=                    ; только чтение! Никакого write
; Monitor через AMI-действие Monitor → разрешено через read=call

; === AMI для администрирования ===
[ami-admin]
secret=admin-ami-very-long-secret
deny=0.0.0.0/0.0.0.0
permit=127.0.0.1/255.255.255.255  ; только localhost
read=all
write=all

Fail2ban для Asterisk SIP

; /etc/fail2ban/jail.local
[asterisk]
enabled  = true
port     = 5060,5061
filter   = asterisk
logpath  = /var/log/asterisk/full
maxretry = 5
bantime  = 3600
findtime = 300

; /etc/fail2ban/filter.d/asterisk.conf
[Definition]
failregex = NOTICE.* <HOST> failed to authenticate
            NOTICE.* Registration from .* failed for <HOST>
            NOTICE.* <HOST>.* No matching peer found

Проверка изоляции

# Тест: PA-колонка не может позвонить на внутренний телефон
# (должна получить отбой, не соединение)
asterisk -rx "channel originate PJSIP/pa-zone-1 extension 101@internal-phones"
# Ожидаемый результат: INVALID EXTENSION или BUSY

# Тест: телефон не может инициировать ChanSpy
# (контекст internal-phones не содержит ChanSpy extensions)
asterisk -rx "channel originate PJSIP/phone-101 extension spy-authorized@internal-phones"
# Ожидаемый результат: INVALID EXTENSION
