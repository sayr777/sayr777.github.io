PostgreSQL: непрерывное резервирование

#!/bin/bash
# backup-postgres.sh — запускать через cron каждые 6 часов

BACKUP_DIR="/backups/postgres"
S3_BUCKET="s3://tenantpass-backups"
DATE=$(date +%Y%m%d_%H%M)

# 1. pg_dump для каждой базы
for DB in tenantpass_config tenantpass_audit tenantpass_tabel eventpass; do
  pg_dump -Fc -d "$DB" -U postgres \
    > "${BACKUP_DIR}/${DB}_${DATE}.dump"
  
  # Шифрование перед отправкой в S3
  gpg --batch --yes --cipher-algo AES256 \
    --passphrase-file /etc/backup/gpg-passphrase \
    --symmetric "${BACKUP_DIR}/${DB}_${DATE}.dump"
  
  # Загрузка в S3 / Яндекс.Облако / MinIO
  aws s3 cp "${BACKUP_DIR}/${DB}_${DATE}.dump.gpg" \
    "${S3_BUCKET}/postgres/${DB}/${DATE}.dump.gpg"
  
  # Удалить локальную незашифрованную копию
  rm "${BACKUP_DIR}/${DB}_${DATE}.dump"
done

# 2. WAL archiving (postgresql.conf)
# archive_mode = on
# archive_command = 'aws s3 cp %p s3://tenantpass-backups/wal/%f'

# postgresql.conf — включить WAL archiving
wal_level = replica
archive_mode = on
archive_command = 'gzip < %p | gpg --batch --passphrase-file /etc/backup/gpg-passphrase --symmetric -o - | aws s3 cp - s3://tenantpass-backups/wal/%f.gz.gpg'
archive_timeout = 300  # принудительный WAL-сегмент каждые 5 мин

Keycloak: экспорт realm

#!/bin/bash
# backup-keycloak.sh

DATE=$(date +%Y%m%d_%H%M)
BACKUP_DIR="/backups/keycloak"

# Экспорт realm через Keycloak Admin CLI
/opt/keycloak/bin/kc.sh export \
  --dir "${BACKUP_DIR}/realm_${DATE}" \
  --users realm_file \
  --realm tenantpass

# Экспорт realm EventPass
/opt/keycloak/bin/kc.sh export \
  --dir "${BACKUP_DIR}/realm_ep_${DATE}" \
  --users realm_file \
  --realm eventpass

# Архив + S3
tar -czf "${BACKUP_DIR}/keycloak_${DATE}.tar.gz" \
  "${BACKUP_DIR}/realm_${DATE}" \
  "${BACKUP_DIR}/realm_ep_${DATE}"

aws s3 cp "${BACKUP_DIR}/keycloak_${DATE}.tar.gz" \
  "s3://tenantpass-backups/keycloak/${DATE}.tar.gz"

# Хранить 90 дней
aws s3 ls s3://tenantpass-backups/keycloak/ | \
  awk '{print $4}' | sort | head -n -90 | \
  xargs -I {} aws s3 rm "s3://tenantpass-backups/keycloak/{}"

Home Assistant: встроенный backup

# configuration.yaml — автоматические backup в Яндекс.Диск / MinIO
# HA Add-on: Samba backup или Google Drive backup

# Через HA automation: ежедневный backup
- alias: "Система: ежедневный backup HA"
  trigger:
    - platform: time
      at: "03:00:00"
  action:
    - service: backup.create
      data:
        name: "daily-{{ now().strftime('%Y%m%d') }}"

# Хранение: local NAS (Samba) + Яндекс.Диск через rclone
# Retention: 30 дней

# rclone sync backup к Яндекс.Диску (cron)
rclone sync /backup/homeassistant/ yandex:HomePass-Backups/ \
  --min-age 0 --max-age 720h  # 30 дней

Frigate NVR: RAID1 + ротация

# frigate.yml
record:
  enabled: true
  retain:
    days: 30           # хранить 30 дней обычные записи
    mode: motion       # только с движением (экономия места)
  events:
    retain:
      default: 90      # события с детекцией — 90 дней
      objects:
        person: 180    # события с людьми — 180 дней

Проверка восстановления (Disaster Recovery drill)

Раз в квартал выполнять тестовое восстановление:

# verify-backups.sh — автоматическая проверка
#!/bin/bash

# 1. Скачать последний backup PostgreSQL
LATEST=$(aws s3 ls s3://tenantpass-backups/postgres/tenantpass_config/ | \
  sort | tail -1 | awk '{print $4}')

aws s3 cp "s3://tenantpass-backups/postgres/tenantpass_config/${LATEST}" /tmp/test.dump.gpg

# 2. Расшифровать и восстановить в тестовую БД
gpg --batch --passphrase-file /etc/backup/gpg-passphrase \
  --decrypt /tmp/test.dump.gpg | \
  pg_restore -d postgres_test_restore -U postgres

# 3. Проверить что данные читаются
RECORD_COUNT=$(psql -U postgres -d postgres_test_restore \
  -t -c "SELECT COUNT(*) FROM access_rules")

if [ "$RECORD_COUNT" -gt 0 ]; then
  echo "✅ Backup verified: ${RECORD_COUNT} records"
else
  echo "❌ Backup FAILED: empty database"
  # Отправить alert
  curl -X POST "$ALERT_WEBHOOK" -d '{"text":"BACKUP VERIFY FAILED!"}'
fi

# 4. Удалить тестовую БД
dropdb -U postgres postgres_test_restore
