Отправка Usage Event при каждом чекине

# eventpass/services/billing.py
import httpx

LAGO_URL = "https://billing.yourcompany.ru/api/v1"
LAGO_KEY = settings.LAGO_API_KEY

async def report_checkin(event_id: str, attendee_id: str, tenant_id: str):
    async with httpx.AsyncClient() as client:
        await client.post(
            f"{LAGO_URL}/events",
            headers={"Authorization": f"Bearer {LAGO_KEY}"},
            json={
                "event": {
                    "transaction_id": f"checkin-{attendee_id}-{event_id}",
                    "external_customer_id": tenant_id,
                    "code": "eventpass_checkins",
                    "timestamp": int(time.time()),
                    "properties": {
                        "event_id": event_id,
                        "attendee_id": attendee_id
                    }
                }
            }
        )

Asterisk SIP — биллинг звонков (CDR)

Каждый звонок записывается в CDR (Call Detail Record). CDR → биллинг.

sequenceDiagram participant USER as Абонент participant AST as Asterisk participant CDR_DB as CDR PostgreSQL participant LAGO as Lago participant INV as Счёт клиенту USER->>AST: Звонок (SIP INVITE) AST->>AST: Разговор USER->>AST: Завершение (BYE) AST->>CDR_DB: INSERT cdr (src, dst, duration, billsec, disposition) CDR_DB->>LAGO: Webhook / batch: usage event\n{minutes: 2.5, tenant: "xyz"} LAGO->>INV: Добавить к счёту за период

CDR → Lago (Python worker)

# billing/cdr_worker.py — запускается каждые 5 минут
import asyncpg, httpx, asyncio
from datetime import datetime, timedelta

async def sync_cdr_to_lago():
    pg = await asyncpg.connect(CDR_DSN)
    
    # Берём необработанные записи
    rows = await pg.fetch("""
        SELECT id, src, dst, billsec, calldate, tenant_id
        FROM cdr
        WHERE billed = false
          AND disposition = 'ANSWERED'
          AND billsec > 0
    """)
    
    events = []
    for row in rows:
        # Округление до 1 минуты вверх (стандарт)
        minutes = max(1, -(-row["billsec"] // 60))
        events.append({
            "transaction_id": f"cdr-{row['id']}",
            "external_customer_id": row["tenant_id"],
            "code": "sip_minutes",
            "timestamp": int(row["calldate"].timestamp()),
            "properties": {
                "src": row["src"],
                "dst": row["dst"],
                "minutes": minutes,
                "billsec": row["billsec"]
            }
        })
    
    async with httpx.AsyncClient() as client:
        for event in events:
            await client.post(
                f"{LAGO_URL}/events",
                headers={"Authorization": f"Bearer {LAGO_KEY}"},
                json={"event": event}
            )
    
    # Пометить как обработанные
    ids = [r["id"] for r in rows]
    await pg.execute("UPDATE cdr SET billed=true WHERE id=ANY($1)", ids)
    await pg.close()

asyncio.run(sync_cdr_to_lago())

Альтернатива: CGRateS — realtime биллинг SIP

YooKassa (Яндекс Касса) — основной российский эквайринг.

Интеграция

# billing/payment/yokassa.py
from yookassa import Configuration, Payment
from yookassa.domain.request import PaymentRequest

Configuration.account_id = settings.YOKASSA_SHOP_ID
Configuration.secret_key  = settings.YOKASSA_SECRET_KEY

def create_payment(
    amount_rub: float,
    description: str,
    invoice_id: str,
    customer_email: str,
    return_url: str
) -> dict:
    payment = Payment.create({
        "amount": {
            "value": f"{amount_rub:.2f}",
            "currency": "RUB"
        },
        "confirmation": {
            "type": "redirect",
            "return_url": return_url
        },
        "description": description,
        "metadata": {
            "invoice_id": invoice_id
        },
        "receipt": {
            "customer": {"email": customer_email},
            "items": [{
                "description": description,
                "quantity": "1",
                "amount": {"value": f"{amount_rub:.2f}", "currency": "RUB"},
                "vat_code": 1,                   # НДС 20%
                "payment_mode": "full_payment",
                "payment_subject": "service"
            }]
        },
        "capture": True
    })
    return {
        "payment_id": payment.id,
        "confirmation_url": payment.confirmation.confirmation_url,
        "status": payment.status
    }

Webhook от YooKassa → Lago

# billing/webhooks/yokassa.py
from fastapi import APIRouter, Request
import hmac, hashlib

router = APIRouter()

@router.post("/webhooks/yokassa")
async def yokassa_webhook(request: Request):
    body = await request.body()
    event = await request.json()
    
    if event["event"] == "payment.succeeded":
        payment = event["object"]
        invoice_id = payment["metadata"]["invoice_id"]
        
        # Пометить счёт как оплаченный в Lago
        await lago_client.post(
            f"/api/v1/invoices/{invoice_id}/payment",
            json={"payment_status": "succeeded",
                  "provider_payment_id": payment["id"]}
        )
        
        # Активировать подписку / открыть доступ
        await activate_subscription(invoice_id)
    
    return {"status": "ok"}

Фискализация — 54-ФЗ (онлайн-касса)

При оплате от физических лиц — обязательный электронный чек.

sequenceDiagram participant CLIENT as Клиент (физлицо) participant APP as TenantPass / EventPass participant YOKA as YooKassa participant ATOL as АТОЛ Онлайн participant OFD as ОФД CLIENT->>APP: Оплата картой APP->>YOKA: Создать платёж с receipt{} YOKA->>YOKA: Списание с карты YOKA->>ATOL: Запрос на фискализацию ATOL->>OFD: Передача фискального документа OFD-->>ATOL: ФПД (фискальный признак) ATOL->>YOKA: Чек сформирован YOKA->>CLIENT: Электронный чек (email / SMS) YOKA-->>APP: Webhook: payment.succeeded

YooKassa сама интегрирована с АТОЛ — при передаче receipt{} в запросе платежа чек формируется автоматически. Для юрлиц (B2B) — чек не нужен, достаточно счёта-фактуры.

Интеграция с 1С:Бухгалтерия

graph LR LAGO["Lago\nсчета / оплаты"] -->|REST API| BRIDGE["1С-мост\n(Python / 1С:Коннектор)"] BRIDGE -->|HTTP + XML| C1["1С:Бухгалтерия 8"] C1 -->|реализации / поступления| REPORT["Бухгалтерская\nотчётность"]
# billing/accounting/c1_bridge.py

import requests

C1_URL = "http://192.168.1.200/base/hs/billing/v1"
C1_AUTH = ("billing_user", "billing_pass")

def post_invoice_to_1c(invoice: dict):
    """Передать счёт на реализацию услуг в 1С"""
    payload = {
        "Номер": invoice["number"],
        "Дата": invoice["issued_at"],
        "Контрагент": invoice["customer"]["legal_name"],
        "ИНН": invoice["customer"]["tin"],
        "Услуги": [
            {
                "Наименование": item["name"],
                "Количество": item["units"],
                "Цена": item["unit_price"],
                "НДС": "20%"
            }
            for item in invoice["fees"]
        ],
        "СуммаИтого": invoice["total_amount"]
    }
    r = requests.post(
        f"{C1_URL}/invoice",
        json=payload,
        auth=C1_AUTH,
        timeout=10
    )
    r.raise_for_status()
    return r.json()["НомерДокумента1С"]

Биллинг сетевого трафика (EventPass WiFi)

Участники получают базовый интернет бесплатно, сверхлимит — платно (или блокировка).

# billing/network/traffic_monitor.py
# Читает Netflow из OpenWrt (через collectd или pmacct)

async def check_guest_quota(mac_address: str, tenant_id: str):
    used_mb = await get_traffic_used(mac_address)   # из InfluxDB / Prometheus
    quota_mb = await get_quota(tenant_id)            # из Lago wallet
    
    if used_mb > quota_mb:
        # Заблокировать в OpenWrt через SSH/API
        await openwrt_client.block_mac(mac_address)
        
        # Показать captive portal с предложением купить дополнительный трафик
        await notify_captive_portal(mac_address, "quota_exceeded")
        
        # Или: автоматическое списание из wallet
        if await lago_client.has_wallet_balance(tenant_id):
            await lago_client.debit_wallet(
                tenant_id,
                amount=100,  # ₽ за 1 ГБ
                description="Extra WiFi 1 GB"
            )
            await openwrt_client.unblock_mac(mac_address)
            await extend_quota(mac_address, 1024)
