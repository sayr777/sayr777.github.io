Интеграция динамических BLE-меток DN·Beacon в TenantPass, HomePass, EventPass

Что такое DN·Beacon

Промышленная BLE-метка на nRF52832 с динамическим ID (меняется каждые 5 мин, AES-128) — защищена от клонирования на аппаратном уровне. IP67, −55…+85°C, до 10 лет от батареи, дальность до 70 м. REST API с JSON.

Ключевые модификации для наших проектов:

ESP32 как BLE-сканер (ESPHome)

ESP32 сканирует BLE-объявления DN·Beacon и публикует в MQTT.

# esphome/ble-scanner-entrance.yaml
esphome:
  name: ble-scanner-entrance
  platform: ESP32
  board: esp32dev

wifi:
  ssid: !secret wifi_ssid
  password: !secret wifi_password

mqtt:
  broker: 192.168.1.100
  topic_prefix: dn/scanner/entrance

esp32_ble_tracker:
  scan_parameters:
    interval: 1100ms
    window: 1100ms
    active: false   # passive scan — не отправляем запросы

ble_presence:
  # Замена ESPresense на DN·Beacon (динамический UUID)
  # Для динамических ID: использовать фильтр по manufacturer data + AES-128
  # Или: использовать DN Gateway API → MQTT bridge

sensor:
  - platform: ble_rssi
    mac_address: FF:FF:FF:FF:FF:FF   # для статических ID (тест)
    name: "DN·Beacon — входная зона — RSSI"
    id: beacon_rssi

# Детекция по iBeacon UUID (меняется каждые 5 мин — нужен DN·Gateway для расшифровки)

Интеграция через DN REST API

DN·Beacon имеет открытый REST API. Наш сервис подписывается на события.

# dn_beacon_bridge.py
# Получает события от DN Gateway → публикует в MQTT / HA

import httpx, asyncio, json
import paho.mqtt.client as mqtt

DN_GATEWAY_URL = "http://192.168.1.150:8080"   # DN Gateway
MQTT_BROKER = "192.168.1.100"

mqtt_client = mqtt.Client()
mqtt_client.connect(MQTT_BROKER)

async def poll_dn_events():
    async with httpx.AsyncClient() as client:
        while True:
            resp = await client.get(f"{DN_GATEWAY_URL}/api/v1/events",
                                    params={"since": last_event_id})
            events = resp.json()
            for event in events:
                process_event(event)
            await asyncio.sleep(0.5)

def process_event(event: dict):
    beacon_id  = event["beacon_id"]   # расшифрованный стабильный ID
    rssi       = event["rssi"]
    event_type = event["type"]        # "enter_zone" | "exit_zone" | "man_down" | "button"
    zone       = event["zone"]
    ts         = event["timestamp"]

    # Публикуем в MQTT
    topic = f"dn/beacon/{beacon_id}/{zone}"
    payload = json.dumps({
        "rssi": rssi,
        "event": event_type,
        "ts": ts
    })
    mqtt_client.publish(topic, payload)

# На входных воротах ESP32 сканирует DN·Pass браслеты участников

async def handle_event_entry(beacon_id: str, zone_id: str, rssi: int):
    if rssi < -75:
        return   # слишком далеко, не засчитывать

    # Проверить: у этого beacon_id есть билет в эту зону?
    ticket = await db.fetchrow("""
        SELECT t.*, z.zone_id
        FROM tickets t
        JOIN beacon_assignments ba ON ba.ticket_id = t.id
        WHERE ba.beacon_id = $1 AND t.event_id = $2
    """, beacon_id, current_event_id)

    if ticket and zone_id in ticket["permitted_zones"]:
        await open_speed_gate(zone_id)
        await mark_entry(ticket["id"], zone_id)
    else:
        await signal_deny(zone_id)

DN·Safe / DN·Man — охрана труда (TenantPass/EventPass)

# dn_safety_monitor.py
# Слушаем события man_down от DN·Man в каске/жилете

async def handle_safety_event(event: dict):
    if event["type"] == "man_down":
        beacon_id = event["beacon_id"]
        worker = await get_worker_by_beacon(beacon_id)

        # Немедленный алерт охраннику и руководителю
        await send_alert(
            title=f"ТРЕВОГА: человек упал! {worker['name']}",
            message=f"Зона: {event['zone']}, координаты: {event['location']}",
            recipients=["security", "manager", "medic"]
        )

        # Открыть ближайшую дверь для доступа медиков
        await open_door(nearest_door_to(event["zone"]))

    elif event["type"] == "enter_hazard_zone":
        beacon_id = event["beacon_id"]
        worker = await get_worker_by_beacon(beacon_id)
        # Проверить: есть ли допуск к опасной зоне?
        if not worker.get("hazard_clearance"):
            await signal_warning(event["zone"])  # звуковой сигнал браслету
            await notify_supervisor(worker, event["zone"])
