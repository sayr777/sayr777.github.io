Топология Keycloak (замечание 14)

Shared vs отдельный Keycloak для каждого проекта

Решение: отдельные реалмы, один кластер

Один кластер Keycloak, три изолированных realm:

graph TD subgraph KC["Keycloak Cluster (2 ноды HA)"] REALM_TP["Realm: tenantpass\nОрганизации = арендаторы\nUsers = сотрудники"] REALM_EP["Realm: eventpass\nОрганизации = мероприятия\nUsers = участники/организаторы"] REALM_MASTER["Realm: master\nТолько admin-операции\nне используется для auth"] DB_KC["PostgreSQL\nKeycloak DB\n(отдельная от app БД)"] end TP_API["TenantPass API"] --> REALM_TP EP_API["EventPass API"] --> REALM_EP ADMIN["Keycloak Admin"] --> REALM_MASTER KC --- DB_KC

Почему один кластер, а не три инстанса

КритерийТри инстансаОдин кластер, три realm
Изоляция данныхПолнаяПолная (realm = изолировано)
Ресурсы3×RAM (3×512 МБ min)1×RAM (1–2 ГБ)
Обновления3 отдельных деплояОдин деплой
Мониторинг3 набора метрикОдин набор
SPOFТри SPOF вместо одногоОдин SPOF → решить через HA
РекомендацияEnterprise с разными физическими зонами✅ Для стартапа/MVP

Высокая доступность Keycloak (2 ноды)

# docker-compose.keycloak-ha.yml
services:
  keycloak-1:
    image: quay.io/keycloak/keycloak:23.0
    environment:
      KC_DB: postgres
      KC_DB_URL: jdbc:postgresql://postgres:5432/keycloak
      KC_DB_USERNAME: keycloak
      KC_DB_PASSWORD: secret
      KC_HOSTNAME: auth.tenantpass.ru
      KC_HTTP_ENABLED: "true"
      KC_PROXY: edge
      KC_CACHE: ispn              # Infinispan distributed cache
      KC_CACHE_STACK: kubernetes  # или 'tcp' для Docker
      JGROUPS_DISCOVERY_PROTOCOL: TCPPING
      JGROUPS_DISCOVERY_INITIAL_HOSTS: "keycloak-1[7600],keycloak-2[7600]"
    command: start
    ports:
      - "8080"
    networks:
      - keycloak_net

  keycloak-2:
    image: quay.io/keycloak/keycloak:23.0
    environment:
      # Те же переменные что и keycloak-1
      JGROUPS_DISCOVERY_INITIAL_HOSTS: "keycloak-1[7600],keycloak-2[7600]"
    command: start
    ports:
      - "8080"
    networks:
      - keycloak_net

  nginx-lb:
    image: nginx:alpine
    volumes:
      - ./nginx-kc.conf:/etc/nginx/conf.d/default.conf
    ports:
      - "443:443"
    depends_on:
      - keycloak-1
      - keycloak-2

  postgres:
    image: postgres:15
    environment:
      POSTGRES_DB: keycloak
      POSTGRES_USER: keycloak
      POSTGRES_PASSWORD: secret

networks:
  keycloak_net:

# nginx-kc.conf — балансировка между двумя нодами
upstream keycloak {
    least_conn;
    server keycloak-1:8080;
    server keycloak-2:8080;
    keepalive 32;
}

server {
    listen 443 ssl;
    server_name auth.tenantpass.ru;

    ssl_certificate     /etc/ssl/certs/tenantpass.crt;
    ssl_certificate_key /etc/ssl/private/tenantpass.key;

    location / {
        proxy_pass http://keycloak;
        proxy_set_header X-Forwarded-For $remote_addr;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header Host $host;
    }
}

Realm-конфигурация

# Создание realm через Keycloak Admin CLI

# TenantPass realm
kcadm.sh create realms -s realm=tenantpass -s enabled=true \
  -s ssoSessionMaxLifespan=28800 \
  -s accessTokenLifespan=300 \
  -s organizations=true     # включить Organizations (multi-tenancy)

# EventPass realm
kcadm.sh create realms -s realm=eventpass -s enabled=true \
  -s ssoSessionMaxLifespan=86400 \  # билет на сутки
  -s accessTokenLifespan=600

HomePass: отдельный Keycloak не нужен

HomePass использует Home Assistant как единственный IdP:

  • Пользователи: только члены семьи (3–5 человек)
  • Гости: временные HA-учётки или Home Assistant People + BLE
  • Keycloak для HomePass избыточен — HA Auth Provider достаточно
# HomePass: HA встроенная аутентификация
# configuration.yaml — без изменений, HA auth встроен

# Для внешнего доступа: Cloudflare Tunnel или VPN (WireGuard)
# Не нужен отдельный OAuth2 сервер

Мониторинг Keycloak

# Keycloak метрики → Prometheus → Grafana
# keycloak env:
KC_METRICS_ENABLED: "true"

# prometheus.yml
scrape_configs:
  - job_name: 'keycloak'
    static_configs:
      - targets: ['keycloak-1:8080', 'keycloak-2:8080']
    metrics_path: /metrics

# Ключевые алерты:
# - keycloak_login_attempts_total{error="true"} > 100/мин → brute force
# - up{job="keycloak"} == 0 → нода упала
# - keycloak_active_sessions > 10000 → проверить утечку сессий

SSO между TenantPass и EventPass

Если нужно единое сквозное SSO — Identity Provider Linking:

TenantPass realm ← Identity Brokering → EventPass realm
  Пользователь TenantPass может логиниться в EventPass через "Login with TenantPass"
  Не обязательно, только если есть общие пользователи (корпоративный EventPass)
