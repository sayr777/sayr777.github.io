ESPHome — DIY реле (ESP32 + 1 канал)

# esphome/relay-single.yaml
esphome:
  name: relay-boiler
  platform: ESP32
  board: esp32dev

wifi:
  ssid: !secret wifi_ssid
  password: !secret wifi_password

api:
  encryption:
    key: !secret api_key

switch:
  - platform: gpio
    name: "Котёл — питание"
    pin:
      number: GPIO26
      inverted: false     # HIGH = включено (для большинства реле-модулей)
    restore_mode: RESTORE_DEFAULT_OFF   # при перезагрузке выключить

# Физическая кнопка управления (опционально)
binary_sensor:
  - platform: gpio
    pin:
      number: GPIO0
      mode: INPUT_PULLUP
      inverted: true
    name: "Кнопка котёл"
    on_press:
      - switch.toggle: relay_boiler

Shelly 1PM — в подрозетник с мониторингом мощности

# Home Assistant: Shelly 1PM автоматически создаёт:
#   switch.shelly_1pm_boiler       — управление
#   sensor.shelly_1pm_boiler_power — мощность (Вт)
#   sensor.shelly_1pm_boiler_energy — потребление (кВт·ч)

# Удобно для:
#   - Отслеживания когда именно работает котёл
#   - Биллинга по потреблению котла
#   - Защиты от перегрева (мощность резко выросла → выключить)

Sonoff ZBMINIL2 — Zigbee без нейтрали

Важная особенность: работает в двухпроводной цепи (без нейтрального провода). Идеально для переделки старой проводки.

Установка в одноклавишный выключатель:
  L (фаза) → ZBMINIL2 → лампа → нейтраль
  Нейтраль не нужна в подрозетнике
  
Физическая кнопка: продолжает работать
HA управление: параллельно через Zigbee

TenantPass — управление нагрузками объекта

Кондиционеры в офисах → Shelly 1PM × N → по расписанию смен
Освещение помещений → Sonoff ZBMINIL2 → по датчику присутствия
Серверная — вентиляция → ESP32 + реле → по температуре
Аварийное освещение → автономное (не через HA)

EventPass (стройплощадка / мероприятие)

Тепловые пушки (обогрев бетона) → ESP32 + контактор → по температуре
Освещение временное → Shelly → по расписанию / закату
Генератор — запуск → реле на стартер → при пропадании сети

Автоматизации HA

# Ночной тариф: бойлер ГВС
- alias: "Ночной тариф: включить бойлер"
  trigger:
    - platform: time
      at: "23:00:00"
  condition:
    - condition: state
      entity_id: input_boolean.night_tariff_enabled
      state: "on"
  action:
    - service: switch.turn_on
      target:
        entity_id: switch.boiler_gvs

- alias: "Ночной тариф: выключить бойлер"
  trigger:
    - platform: time
      at: "07:00:00"
  action:
    - service: switch.turn_off
      target:
        entity_id: switch.boiler_gvs

# Никого нет дома → выключить всё лишнее
- alias: "Уход: экономия энергии"
  trigger:
    - platform: state
      entity_id: group.family_presence
      to: "not_home"
  action:
    - service: switch.turn_off
      target:
        entity_id:
          - switch.warm_floor_living_room
          - switch.warm_floor_bathroom
          - switch.light_outdoor
    - service: climate.set_temperature
      target:
        entity_id: climate.house_thermostat
      data:
        temperature: 16       # режим отсутствия
