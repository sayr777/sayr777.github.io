Изоляция данных СКУД / СУРВ (замечание 3)

TenantPass — разделение данных по функциям и требованиям

Проблема

Одна PostgreSQL для всего: конфигурация СКУД, события проходов, табель СУРВ. Риски:

  • Rollback схемы СКУД затирает данные табеля
  • ФЗ-152: персональные данные и события прохода требуют отдельных политик доступа
  • Трудовой кодекс: табель хранится 5 лет, события СКУД — 1 год

Решение: три отдельные базы данных

graph TD subgraph DB_CLUSTER["PostgreSQL (один кластер, разные БД)"] DB_CONFIG["tenantpass_config\n(конфигурация: двери, зоны, расписания)"] DB_AUDIT["tenantpass_audit\n(события прохода — append-only)"] DB_TABEL["tenantpass_tabel\n(табель СУРВ — юридически значимые)"] end subgraph APPS["Приложения"] API_GATEWAY["API Gateway"] PEP["PEP Agent"] SYRV["СУРВ Сервис"] HR["HR-интерфейс"] end API_GATEWAY -->|read/write| DB_CONFIG PEP -->|write events| DB_AUDIT SYRV -->|read events| DB_AUDIT SYRV -->|write/read tabel| DB_TABEL HR -->|read tabel| DB_TABEL

Схемы баз данных

tenantpass_config — конфигурация

-- Сбрасываемая/изменяемая, нет ФЗ-152 рисков
CREATE TABLE doors (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id   UUID NOT NULL,
    name        TEXT NOT NULL,
    fail_mode   TEXT NOT NULL DEFAULT 'fail_open',
    zone_id     UUID REFERENCES zones(id)
);

CREATE TABLE access_rules (
    id          UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    door_id     UUID REFERENCES doors(id),
    subject_id  UUID NOT NULL,   -- user or group in Keycloak
    schedule_id UUID REFERENCES schedules(id),
    valid_from  TIMESTAMPTZ,
    valid_until TIMESTAMPTZ
);

tenantpass_audit — события СКУД

-- Append-only: никогда не UPDATE/DELETE
-- Retention: 1 год (конфигурируется через pg_partman)
CREATE TABLE access_events (
    id          UUID DEFAULT gen_random_uuid(),
    tenant_id   UUID NOT NULL,
    door_id     UUID NOT NULL,
    person_id   UUID NOT NULL,       -- FK к Keycloak user_id
    card_id     TEXT NOT NULL,       -- хэш карты (SHA-256), не сырые данные
    event_type  TEXT NOT NULL,       -- 'granted' | 'denied' | 'forced_open' | 'tailgate'
    source      TEXT NOT NULL,       -- 'keycloak' | 'cache_hit' | 'cache_stale' | 'offline_default'
    occurred_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
    metadata    JSONB
) PARTITION BY RANGE (occurred_at);

-- Автопартиции по месяцу (pg_partman)
SELECT partman.create_parent(
    p_parent_table => 'access_events',
    p_control => 'occurred_at',
    p_type => 'range',
    p_interval => 'monthly',
    p_retention => '12 months',
    p_retention_keep_table => false
);

-- ВАЖНО: права — только INSERT, без UPDATE/DELETE
REVOKE UPDATE, DELETE ON access_events FROM pep_agent_role;

tenantpass_tabel — табель СУРВ

-- Retention: 5 лет (ТК РФ ст.283)
CREATE TABLE tabel_records (
    id              UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    tenant_id       UUID NOT NULL,
    employee_id     UUID NOT NULL,
    record_date     DATE NOT NULL,
    tabel_code      TEXT NOT NULL,
    worked_minutes  INTEGER,
    first_in        TIME,
    last_out        TIME,
    correction_by   UUID,               -- кто корректировал
    correction_at   TIMESTAMPTZ,
    correction_note TEXT,
    source          TEXT DEFAULT 'auto', -- 'auto' | 'manual_hr' | 'manual_manager'
    UNIQUE(employee_id, record_date)
);

-- Партиции по году
SELECT partman.create_parent(
    p_parent_table => 'tabel_records',
    p_control => 'record_date',
    p_type => 'range',
    p_interval => 'yearly',
    p_retention => '5 years',
    p_retention_keep_table => true   -- архивировать, не удалять
);

Политики доступа (Row-Level Security)

-- tenantpass_audit: тенант видит только свои события
ALTER TABLE access_events ENABLE ROW LEVEL SECURITY;

CREATE POLICY tenant_isolation ON access_events
    USING (tenant_id = current_setting('app.current_tenant')::UUID);

-- tenantpass_tabel: HR видит только своих сотрудников
ALTER TABLE tabel_records ENABLE ROW LEVEL SECURITY;

CREATE POLICY hr_access ON tabel_records
    USING (tenant_id = current_setting('app.current_tenant')::UUID);

ФЗ-152: что является персональными данными

ПолеПД?ХранениеШифрование
card_id (сырой номер)ДаХранить хэш SHA-256—
person_id (UUID Keycloak)ПсевдонимOK—
ФИОДаТолько в Keycloak, не в audit—
Биометрия (фото лица)СпецкатегорияТолько с согласия, отдельноAES-256
ГеолокацияДаТолько при явном согласии—
# pep/event_writer.py — никогда не пишем сырые ПД в audit
def write_access_event(person_id: str, raw_card_id: str, door_id: str, permit: bool):
    import hashlib
    card_hash = hashlib.sha256(raw_card_id.encode()).hexdigest()[:16]  # truncate — не reversible
    db_audit.execute("""
        INSERT INTO access_events (person_id, card_id, door_id, event_type, occurred_at)
        VALUES ($1, $2, $3, $4, NOW())
    """, person_id, card_hash, door_id, "granted" if permit else "denied")

# Инструмент: Alembic с отдельными env.py на каждую БД

alembic -c alembic_config.ini upgrade head    # config DB
alembic -c alembic_audit.ini  upgrade head    # audit DB (только additive!)
alembic -c alembic_tabel.ini  upgrade head    # tabel DB

# Правило: audit и tabel DB — только additive migrations
# (ADD COLUMN, CREATE TABLE — да; DROP TABLE, ALTER TYPE — нет)
