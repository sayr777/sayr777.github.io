Система управления рабочим временем (СУРВ)

TenantPass Corporate — учёт рабочего времени, табель, интеграции

Обзор

СУРВ — вторая ключевая функция TenantPass (рядом со СКУД). Собирает события из СКУД (вход/выход через турникет), формирует табель рабочего времени, интегрируется с 1С:ЗУП.

Табель рабочего времени

Формат совместим с унифицированными формами Т-12 / Т-13 (Постановление Госкомстата РФ №1).

Обозначения:
  Я   — рабочий день (явка)
  В   — выходной
  ОТ  — отпуск
  К   — командировка
  Б   — больничный
  НН  — неявка по невыясненным причинам (опоздание > N мин)
  С   — сверхурочные
  ОЖ  — отпуск по уходу за ребёнком
  Р   — отпуск по беременности и родам

# tabel_generator.py — формирование табеля по событиям СКУД
from dataclasses import dataclass
from datetime import date, datetime, timedelta
from typing import Optional
import json

@dataclass
class AttendanceEvent:
    employee_id: str
    timestamp: datetime
    event_type: str   # "in" | "out"
    door_id: str

@dataclass
class DayRecord:
    date: date
    first_in: Optional[datetime]
    last_out: Optional[datetime]
    scheduled_start: datetime
    scheduled_end: datetime

    @property
    def worked_minutes(self) -> int:
        if not self.first_in or not self.last_out:
            return 0
        return int((self.last_out - self.first_in).total_seconds() / 60)

    @property
    def tabel_code(self) -> str:
        if not self.first_in:
            return "НН"
        late_minutes = (self.first_in - self.scheduled_start).total_seconds() / 60
        if late_minutes > 60:
            return "НН"
        if self.worked_minutes > (self.scheduled_minutes + 30):
            return "С"
        return "Я"

    @property
    def scheduled_minutes(self) -> int:
        return int((self.scheduled_end - self.scheduled_start).total_seconds() / 60) - 60  # - обед

def build_tabel(employee_id: str, month: int, year: int,
                events: list[AttendanceEvent], schedule) -> list[DayRecord]:
    records = []
    for day in range(1, 32):
        try:
            d = date(year, month, day)
        except ValueError:
            break
        day_events = [e for e in events if e.timestamp.date() == d]
        ins  = [e for e in day_events if e.event_type == "in"]
        outs = [e for e in day_events if e.event_type == "out"]
        shift = schedule.get_shift(employee_id, d)
        records.append(DayRecord(
            date=d,
            first_in=min(ins, key=lambda e: e.timestamp).timestamp if ins else None,
            last_out=max(outs, key=lambda e: e.timestamp).timestamp if outs else None,
            scheduled_start=datetime.combine(d, shift.start) if shift else None,
            scheduled_end=datetime.combine(d, shift.end) if shift else None,
        ))
    return records

Интеграция с 1С:ЗУП

Ежедневная или ежемесячная выгрузка табеля в 1С для расчёта зарплаты.

# zup_export.py — генерация XML для 1С:ЗУП

import xml.etree.ElementTree as ET
from datetime import date

def generate_tabel_xml(records: list[DayRecord], org_id: str) -> str:
    root = ET.Element("TabelXML")
    root.set("xmlns", "urn:1c.ru:ZUP.Tabel.v1")
    root.set("OrganizationID", org_id)
    root.set("Month", records[0].date.strftime("%Y-%m"))

    for rec in records:
        row = ET.SubElement(root, "Row")
        ET.SubElement(row, "EmployeeID").text = rec.employee_id
        ET.SubElement(row, "Date").text = str(rec.date)
        ET.SubElement(row, "Code").text = rec.tabel_code
        ET.SubElement(row, "Hours").text = str(round(rec.worked_minutes / 60, 1))
        if rec.first_in:
            ET.SubElement(row, "TimeIn").text  = rec.first_in.strftime("%H:%M")
            ET.SubElement(row, "TimeOut").text = rec.last_out.strftime("%H:%M") if rec.last_out else ""

    return ET.tostring(root, encoding="unicode", xml_declaration=True)

# Отправка в 1С через HTTP
import httpx

async def push_to_1c(xml_data: str, c1_url: str, token: str):
    async with httpx.AsyncClient() as client:
        resp = await client.post(
            f"{c1_url}/TabelImport",
            content=xml_data,
            headers={"Content-Type": "application/xml", "Authorization": f"Bearer {token}"},
        )
    return resp.status_code == 200

Ключевые метрики на дашборде:

  • Средний процент присутствия по отделу / за период
  • Топ по опозданиям / прогулам
  • Количество сверхурочных часов
  • Реалтайм: кто сейчас в офисе
  • Тепловая карта: загруженность по дням недели и часам
